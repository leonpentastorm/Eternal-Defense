    // The block outline: counted after every other listener, canceled or not, by phase of the print
    static int phase=-1;static final int[] DRAWN=new int[5],HIDDEN=new int[5];
    @SubscribeEvent(priority=EventPriority.LOWEST,receiveCanceled=true) public static void outline(net.minecraftforge.client.event.RenderHighlightEvent.Block e){
        if(phase<0||phase>4)return;
        // The print's own frames end with its last tick (the doors shut); a frame after that, before the server's end arrives, counts as after
        int ph=phase;if(ph==1){var tl=clientTimeline();if(tl!=null&&clientElapsed()>=tl.end())ph=2;}
        if(e.isCanceled())HIDDEN[ph]++;else DRAWN[ph]++;
        if(!e.isCanceled()&&(ph==1||ph==3))note("OUTLINE DRAWN phase {} t={} on {} {}",ph,clientElapsed(),e.getTarget().getBlockPos().toShortString(),BuiltInRegistries.BLOCK.getKey(mc().level.getBlockState(e.getTarget().getBlockPos()).getBlock()));
    }

    @SubscribeEvent public static void onTick(TickEvent.ClientTickEvent e){
        if(e.phase!=TickEvent.Phase.END)return;var mc=mc();
        if(!created){
            if(mc.screen instanceof TitleScreen&&mc.getOverlay()==null&&++tick>60){
                created=true;var settings=new LevelSettings("qa_r113",GameType.SURVIVAL,false,Difficulty.PEACEFUL,true,new GameRules(),WorldDataConfiguration.DEFAULT);
                mc.createWorldOpenFlows().createFreshLevel("qa_r113",settings,new WorldOptions(42L,false,false),access->access.registryOrThrow(Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions());
            }
            return;
        }
        if(mc.player==null||mc.level==null||mc.getSingleplayerServer()==null)return;
        if(!built){if(++tick<200)return;built=true;sx=(int)Math.floor(mc.player.getX());sy=(int)Math.floor(mc.player.getY());sz=(int)Math.floor(mc.player.getZ());places();script();}
        if(holding)return;
        if(wait>0){wait--;return;}
        var head=STEPS.peek();
        if(head!=null&&head.length>3){boolean ok;try{ok=((java.util.function.BooleanSupplier)head[3]).getAsBoolean();}catch(Throwable t){ok=false;}int[] rem=(int[])head[4];if(!ok&&rem[0]-->0){wait=0;return;}if(!ok)log("QA_TIMEOUT {}",head[2]);}
        var step=STEPS.poll();
        if(step==null){log("QA_DONE pass={} fail={}",pass,fail);mc.stop();return;}
        log("QA_STEP {}",step[2]);
        try{((Runnable)step[1]).run();}catch(Throwable t){fail++;com.mojang.logging.LogUtils.getLogger().error("QA_STEP_ERROR {}",step[2],t);}
        wait=(int)step[0];
    }

            check("the kitchen and support rows are never 'empty'",srvGet(()->WorkshopFabrication.ITEMS.stream().filter(id->!TABLES.contains(id)).noneMatch(id->WorkshopFabrication.empty(sp(),id))),"");
        },"which tables are empty");
        step(5,()->{
            String armor=srvGet(()->WorkshopFabrication.buy(sp(),"armor_platform")),gun=srvGet(()->WorkshopFabrication.buy(sp(),"gun_platform"));note("BUY armor: {} | gun: {}",armor,gun);
            check("buying the armor table is refused: nothing installed adds anything to make there",armor.contains("no installed mod adds anything to make there"),armor);

        },"buy: refused for the empty table only");
        step(5,()->srv(()->BeaconNetwork.open(sp(),"menu","")),"open the beacon");
        until("beacon screen",60,()->mc().screen instanceof BeaconClient.ControlScreen,()->{});
        step(10,()->{var s=(BeaconClient.ControlScreen)mc().screen;for(var c:s.children())if(c instanceof net.minecraft.client.gui.components.Button b&&b.getMessage().getString().equals(Ui.t("tab.fabrication").getString())){b.onPress();break;}},"the Workshop Fabrication tab");
        step(10,()->{
            var s=(BeaconClient.ControlScreen)mc().screen;var armor=fabRow("armor_platform");
            check("the tab opens",s.tab==2&&s.fabricationButtons.size()==WorkshopFabrication.ITEMS.size(),s.tab+" "+s.fabricationButtons.size());
            check("the client is told the armor table is empty",armor!=null&&armor.getBoolean("empty"),String.valueOf(armor));
            check("and the other three tables are not",TABLES.stream().filter(t->!t.equals("armor_platform")).allMatch(t->{var r=fabRow(t);return r!=null&&!r.getBoolean("empty");}),"");
            int i=WorkshopFabrication.ITEMS.indexOf("armor_platform");var b=s.fabricationButtons.get(i);
            check("its Make button is off and not marked as short of materials",!b.active&&!b.warning,"active="+b.active+" warning="+b.warning);
            var c=cell(s,i);hover(c[0]+10,c[1]+c[3]/2.0);
        },"the armor card");
        step(8,()->shot("r113_fabrication_armor_empty"),"picture: the armor card and its tooltip");
        step(3,()->{var s=(BeaconClient.ControlScreen)mc().screen;var c=cell(s,WorkshopFabrication.ITEMS.indexOf("gun_platform"));hover(c[0]+10,c[1]+c[3]/2.0);},"hover the weapon table");
        step(8,()->shot("r113_fabrication_gun"),"picture: a table with things to make");
        step(5,()->{((BeaconClient.PanelScreen)mc().screen).message="";BeaconNetwork.action("fabricate:armor_platform","");},"send the order anyway (as a modified client would)");
        step(15,()->{var m=mc().screen instanceof BeaconClient.PanelScreen p?p.message:"no screen";
            check("the server answers on the beacon screen that nothing installed adds anything to make there",m.contains("no installed mod adds anything to make there"),m);
            check("still no armor table in the pack",srvGet(()->inPack(WorkshopFabrication.item("armor_platform"))==0),"");},"refused");
        step(5,()->mc().setScreen(null),"close");

        // 3. The Loadout Station: no block outline while the print plays
        step(20,()->{clearMessages();srv(()->{sp().getInventory().setItem(0,st(EternalDefense.LOADOUT_STATION_ITEM.get(),1));sp().getInventory().selected=0;});mc().player.getInventory().selected=0;tp(S.getX()+.5,sy,S.getZ()-4.5,0,30);},"hold the station in front of its spot");
        step(5,()->click(S.below(2),Direction.UP),"place it (a real right-click on the ground)");
        step(20,()->check("the station stands (27 cells)",srvGet(()->{int ok=0;for(var p:LoadoutStation.cells(S))if(LoadoutStation.isStation(sl().getBlockState(p))&&LoadoutStation.anchor(sl(),p).equals(S))ok++;return ok;})==27,""),"placed");
        step(5,()->srv(()->{var m=LoadoutStation.machine(sl(),S);var coin=LoadoutCoin.create(kit());coin.setCount(3);m.shelf.setItem(0,coin);
            var inv=sp().getInventory();for(int i=0;i<36;i++)inv.setItem(i,ItemStack.EMPTY);for(int i=0;i<4;i++)inv.armor.set(i,ItemStack.EMPTY);inv.offhand.set(0,ItemStack.EMPTY);
            inv.items.set(3,st(Items.APPLE,5));inv.armor.set(3,st(Items.IRON_HELMET,1));inv.setChanged();}),"three coins in slot 1; plain gear");
        for(boolean cinematic:new boolean[]{true,false}){String tag=cinematic?"camera shots on":"camera shots off";
            step(5,()->{DefenseClientConfig.set(DefenseClientConfig.LOADOUT_CINEMATIC,cinematic);readyNow();face();resetOutline();},tag+": in front, looking at the doors");
            step(5,()->phase=0,"count outlines before");
            step(30,()->{},"look at the station for 1.5 s");
            step(1,()->{phase=-1;check(tag+": before the print the outline is drawn on the station as usual",DRAWN[0]>0&&HIDDEN[0]==0,outlines());},"before");
            step(2,QaRelease113::quickPrint,tag+": sneak + right-click the doors");
            until(tag+": the client sees the print",60,()->clientElapsed()>=0,()->{phase=1;check(tag+": the print starts",printingOnServer(),messages());});
            until(tag+": mid print",400,()->{var tl=clientTimeline();return tl!=null&&clientElapsed()>=tl.cycleStart()+24;},()->shot(cinematic?"r113_print_camera":"r113_print_own_eyes"));
            until(tag+": the print ends",900,()->!printingOnServer()&&clientElapsed()<0,()->phase=2);
            step(2,QaRelease113::face,"back in front of the station");
            step(30,()->{},"look at it for 1.5 s");
            step(1,()->{phase=-1;note("OUTLINE {}: {}",tag,outlines());
                check(tag+": while the print plays no block outline is drawn",DRAWN[1]==0,outlines());
                check(tag+": after it the outline is back",DRAWN[2]>0,outlines());
                check(tag+": the kit was printed (chainmail chestplate worn)",srvGet(()->sp().getInventory().armor.get(2).is(Items.CHAINMAIL_CHESTPLATE)),inventory());},"after");
        }
        step(2,()->DefenseClientConfig.set(DefenseClientConfig.LOADOUT_CINEMATIC,true),"camera shots back on");
        // Another player's print: this player, aside, sees no outline on the station; elsewhere as usual
        step(5,()->{resetOutline();stand(S.getX()+3.5,sy,S.getZ()-3.5,S.getX()+1.5,sy+1.4,S.getZ()-1);},"aside, looking at the station's front");
        step(2,()->{otherPrint(true);phase=3;check("this client sees another player's print playing (not its own)",clientStation().occupant!=null&&LoadoutClient.held()==null,"");},"another player's print, as the server syncs it to this client");
        step(40,()->{},"watch it for 2 s");
        step(1,()->{phase=4;stand(S.getX()+3.5,sy,S.getZ()-3.5,S.getX()+6.5,sy-.5,S.getZ()-3.5);},"look at the ground beside");
        step(20,()->{},"1 s");
        step(1,()->{phase=-1;note("OUTLINE another's print: {}",outlines());
            check("another player's print: no outline on the station while it plays",DRAWN[3]==0&&HIDDEN[3]>0,outlines());
            check("another player's print: other blocks keep their outline",DRAWN[4]>0&&HIDDEN[4]==0,outlines());},"bystander");
        step(5,()->otherPrint(false),"that print over");

        // 4. The replaced models: none missing, each from the front, the front left and behind
        subjects();
        step(10,()->srv(()->{for(var s:SUBJECTS)sl().setBlock(s.pos(),s.state().get(),3);}),"every replaced model in a row, facing north");
        step(60,()->{srv(()->sp().setGameMode(GameType.SPECTATOR));mc().options.hideGui=true;view(ROW.getX()+20,sy+3,ROW.getZ()-12,ROW.getX()+20,sy,ROW.getZ());},"a spectator over the row (the chunks load)");
        step(2,()->{
            var models=mc().getModelManager();var missing=models.getMissingModel();
            var bad=new ArrayList<String>();int states=0;
            for(var e:BuiltInRegistries.BLOCK.entrySet()){if(!e.getKey().location().getNamespace().equals(EternalDefense.ID))continue;
                for(var st:e.getValue().getStateDefinition().getPossibleStates()){if(st.getRenderShape()!=RenderShape.MODEL)continue;states++;if(mc().getBlockRenderer().getBlockModel(st)==missing)bad.add(st.toString());}}
            check("every block state of the mod has its model ("+states+" states)",bad.isEmpty(),String.join(", ",bad.subList(0,Math.min(12,bad.size()))));
            var badItems=new ArrayList<String>();int items=0;
            for(var e:BuiltInRegistries.ITEM.entrySet()){if(!e.getKey().location().getNamespace().equals(EternalDefense.ID))continue;items++;
                if(mc().getItemRenderer().getItemModelShaper().getItemModel(new ItemStack(e.getValue()))==missing)badItems.add(e.getKey().location().getPath());}

            var parts=new ArrayList<String>();int n=0;
            for(var entry:Animations.PARTS.entrySet())for(var p:entry.getValue()){n++;if(models.getModel(Animations.part(entry.getKey(),p))==missing)parts.add(entry.getKey()+"_"+p);}

            var placed=new ArrayList<String>();for(var s:SUBJECTS)if(!mc().level.getBlockState(s.pos()).is(s.state().get().getBlock()))placed.add(s.name());
            check("all "+SUBJECTS.size()+" replaced models stand in the row",placed.isEmpty(),String.join(", ",placed));
        },"models");
        for(var s:SUBJECTS)for(int angle:new int[]{0,35,180}){
            step(4,()->around(s,angle),s.name()+" at "+angle);
            step(14,()->shot("r113_model_"+s.name()+"_"+angle),"picture");
        }
        // The Loadout Station's new model, closed, from the same three sides
        var station=new Subject("loadout_station",null,S,.5,.6,.5,6.4);
        for(int angle:new int[]{0,35,180}){step(4,()->around(station,angle),"loadout station at "+angle);step(14,()->shot("r113_model_loadout_station_"+angle),"picture");}
        step(4,()->{mc().options.hideGui=false;srv(()->sp().setGameMode(GameType.SURVIVAL));tp(sx+.5,sy,sz+.5,0,0);},"HUD back; survival");

            step(10,()->{
                var screen=(net.minecraftforge.client.gui.ModListScreen)mc().screen;var list=(net.minecraftforge.client.gui.widget.ModListWidget)field(screen,"modList");

                check(mod+": the mod list shows its logo, 480 x 160",path!=null&&dims!=null&&dims.width==480&&dims.height==160,path+" "+(dims==null?"":dims.width+"x"+dims.height));
                shot("r113_modlist_"+mod);},"logo of "+mod);
        }
        step(5,()->mc().setScreen(null),"close");
    }
