    static BlockState with(BlockState s,String property,Object value){
        for(Property p:s.getProperties())if(p.getName().equals(property)){var v=p.getValue(value.toString().toLowerCase(Locale.ROOT));if(v.isPresent())return s.setValue(p,(Comparable)v.get());}
        log("QA_NOTE no property {}={} on {}",property,value,s);return s;
    }

    static int total(IItemHandler h){int n=0;if(h!=null)for(int i=0;i<h.getSlots();i++)n+=h.getStackInSlot(i).getCount();return n;}

    @SubscribeEvent public static void onTick(TickEvent.ClientTickEvent e){
        if(e.phase!=TickEvent.Phase.END)return;var mc=mc();
        if(!created){
            if(mc.screen instanceof TitleScreen&&mc.getOverlay()==null&&++tick>60){
                created=true;var settings=new LevelSettings("qa_hub_create",GameType.CREATIVE,false,Difficulty.PEACEFUL,true,new GameRules(),WorldDataConfiguration.DEFAULT);
                mc.createWorldOpenFlows().createFreshLevel("qa_hub_create",settings,new WorldOptions(41L,false,false),access->access.registryOrThrow(Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions());
            }
            return;
        }
        if(mc.player==null||mc.level==null||mc.getSingleplayerServer()==null)return;
        if(!built){if(++tick<200)return;built=true;sx=(int)Math.floor(mc.player.getX());sy=(int)Math.floor(mc.player.getY());sz=(int)Math.floor(mc.player.getZ());places();script();}
        if(holding)return;
        if(wait>0){wait--;return;}
        var head=STEPS.peek();
        if(head!=null&&head.length>3){boolean ok;try{ok=((java.util.function.BooleanSupplier)head[3]).getAsBoolean();}catch(Throwable t){ok=false;}int[] rem=(int[])head[4];if(!ok&&rem[0]-->0){wait=3;return;}if(!ok)log("QA_TIMEOUT {}",head[2]);}
        var step=STEPS.poll();
        if(step==null){log("QA_DONE pass={} fail={}",pass,fail);mc.stop();return;}
        log("QA_STEP {}",step[2]);
        try{((Runnable)step[1]).run();}catch(Throwable t){fail++;com.mojang.logging.LogUtils.getLogger().error("QA_STEP_ERROR {}",step[2],t);}
        wait=(int)step[0];
    }

    /** A mechanical arm's interaction points, as Create saves them: {type id, position, TAKE or DEPOSIT}. */
    static void arm(BlockPos arm,List<Object[]> points){
        var be=sl().getBlockEntity(arm);if(be==null){log("QA_NOTE no arm at {}",arm);return;}
        var list=new ListTag();
        for(var p:points){var t=new CompoundTag();t.putString("Type",(String)p[0]);t.put("Pos",NbtUtils.writeBlockPos(((BlockPos)p[1]).subtract(arm)));t.putString("Mode",(String)p[2]);list.add(t);}
        var tag=be.saveWithoutMetadata();tag.put("InteractionPoints",list);be.load(tag);be.setChanged();
    }
