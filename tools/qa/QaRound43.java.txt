    static final List<String> SOUNDS=Collections.synchronizedList(new ArrayList<>());
    @SubscribeEvent public static void sound(net.minecraftforge.client.event.sound.PlaySoundEvent e){
        var s=e.getOriginalSound();if(s==null||!s.getLocation().getNamespace().equals(ArsenalBeacon.ID))return;

    static long sounds(String prefix){synchronized(SOUNDS){return SOUNDS.stream().filter(s->s.startsWith(prefix)).count();}}
    /**
     * The container has no audio device, so the client's sound engine never starts and plays nothing (no PlaySoundEvent). What the
     * server sends is counted instead: every sound packet of the mod that reaches the client's connection (a handler in front of the
     * connection's own, in the integrated server's local channel the packets arrive as objects).
     */
    static boolean hooked;
    static void hookSounds(){
        if(hooked||mc().getConnection()==null)return;hooked=true;
        mc().getConnection().getConnection().channel().pipeline().addBefore("packet_handler","qa_sounds",new io.netty.channel.ChannelInboundHandlerAdapter(){
            @Override public void channelRead(io.netty.channel.ChannelHandlerContext ctx,Object msg)throws Exception{
                ResourceLocation id=null;
                if(msg instanceof net.minecraft.network.protocol.game.ClientboundSoundPacket p)id=p.getSound().value().getLocation();
                else if(msg instanceof net.minecraft.network.protocol.game.ClientboundSoundEntityPacket p)id=p.getSound().value().getLocation();
                if(id!=null&&id.getNamespace().equals(ArsenalBeacon.ID)){SOUNDS.add(id.getPath());log("QA_SOUND_PACKET {}",id);}
                super.channelRead(ctx,msg);
            }
        });
        log("QA_NOTE sound packets are counted at the client's connection");
    }

        log("QA_STEP {}",step[2]);
        try{((Runnable)step[1]).run();}catch(Throwable t){fail++;com.mojang.logging.LogUtils.getLogger().error("QA_STEP_ERROR {}",step[2],t);}
        wait=(int)step[0];
    }

        until("the gate screen opens",100,()->gate()!=null,()->check("the Receiving Gate's screen opens",gate()!=null,String.valueOf(mc().screen)));

        // ---- the mine's picker ----------------------------------------------------------------------------------------------------------------------
        step(10,()->{var s=gate();Ui.UiButton choose=null;int k=0;if(s!=null)for(var c:s.children())if(c instanceof Ui.UiButton b&&b.visible&&b.getMessage().getString().equals("Choose")&&k++==0)choose=b;
            if(choose!=null)hover(choose.getX()+choose.getWidth()/2.0,choose.getY()+choose.getHeight()/2.0);
            check("the mine's Choose tip names clay and the stones",Ui.t("gate.slot.choose.tip.mine").getString().contains("clay, or a stone for building: calcite, dripstone, red sandstone, blackstone or one of Create's stones"),Ui.t("gate.slot.choose.tip.mine").getString());},"point at the mine's Choose button");
        step(30,()->shot("r43_03_choose_tip"),"its tooltip");
        step(5,()->{hover(1,1);pressNth("Choose",0);},"Choose for the mine");
        step(10,()->{var s=gate();int buttons=s==null?0:s.pickButtons.size();
            check("the mine's grid: 18 items, every one a real item on the client and the server",buttons==RouteRules.MINE_LOOT.size()
                &&RouteRules.MINE_LOOT.stream().allMatch(l->!GateClient.icon(l.id()).isEmpty())&&srvGet(()->RouteRules.MINE_LOOT.stream().allMatch(l->BuiltInRegistries.ITEM.containsKey(new ResourceLocation(l.id())))),
                "buttons "+buttons+" of "+RouteRules.MINE_LOOT.size());
            var b=pick("minecraft:clay");if(b!=null)hover(b.getX()+9,b.getY()+9);},"the grid; point at clay");
        step(30,()->shot("r43_04_mine_picker_clay"),"the grid with clay's tooltip");
        step(5,()->{var b=pick("minecraft:clay");if(b!=null)b.onPress();hover(1,1);},"choose clay");
        step(25,()->{var f=fob();check("the mine now sends clay (the server's choice and the screen's)",f!=null&&f.mine==index("minecraft:clay")&&gateSlot(1).getString("item").equals("minecraft:clay"),f==null?"":f.mine+" "+gateSlot(1).getString("item"));
            note=""+gateSlot(1).getDouble("perTrip");
            srv(()->{var d=FobData.get(sl());d.fobs.values().iterator().next().nextPassive=d.onlineTicks+20;});},"chosen; a delivery is due in a second");
        until("a delivery with clay",200,()->stored("minecraft:clay")>0,()->{var f=fob();int clay=stored("minecraft:clay");
            check("the route delivers clay into the gate's storage",clay>0&&f!=null&&"passive".equals(f.lastKind)&&f.lastItems==clay,"clay "+clay+", last "+(f==null?"":f.lastKind+" "+f.lastItems)+", per trip on the screen "+note);});
        step(25,()->{pressNth("Choose",0);},"Choose for the mine again");
        step(10,()->{var b=pick("create:asurine");if(b!=null)hover(b.getX()+9,b.getY()+9);},"point at asurine");
        step(30,()->shot("r43_05_mine_picker_asurine"),"asurine's tooltip");
        step(5,()->{var b=pick("create:asurine");if(b!=null)b.onPress();hover(1,1);},"choose asurine");
        step(25,()->{var f=fob();double clayPer=Double.parseDouble(note),stonePer=gateSlot(1).getDouble("perTrip");
            check("the mine now sends asurine",f!=null&&f.mine==index("create:asurine")&&gateSlot(1).getString("item").equals("create:asurine"),f==null?"":f.mine+" "+gateSlot(1).getString("item"));
            check("a stone comes twice as many a trip as clay (worth half)",clayPer>0&&Math.abs(stonePer/clayPer-2)<1e-6,"clay "+clayPer+" a trip, asurine "+stonePer);
            srv(()->{var d=FobData.get(sl());d.fobs.values().iterator().next().nextPassive=d.onlineTicks+20;});},"chosen; another delivery");
        until("a delivery with asurine",200,()->stored("create:asurine")>0,()->{var f=fob();int stone=stored("create:asurine");
            check("the route delivers asurine into the gate's storage",stone>0&&f!=null&&f.lastItems==stone,"asurine "+stone+", last "+(f==null?"":f.lastKind+" "+f.lastItems));});
        step(45,()->{var s=gate();int stacks=0;if(s!=null)for(int i=0;i<FobData.DEPOT_SLOTS;i++){var it=s.getMenu().slots.get(i).getItem();if(it.is(item("minecraft:clay"))||it.is(item("create:asurine")))stacks++;}
            check("the screen's storage shows the clay and the asurine",stacks>=2,"stacks "+stacks);shot("r43_06_gate_storage");},"the storage on the screen");
        step(5,()->{mc().setScreen(null);srv(()->sp().closeContainer());},"close");
    }
