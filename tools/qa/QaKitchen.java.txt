        if(e.phase!=TickEvent.Phase.END)return;var mc=mc();
        if(!created){
            if(mc.screen instanceof TitleScreen&&mc.getOverlay()==null&&++tick>60){
                created=true;
                var settings=new LevelSettings("qa_kitchen",GameType.CREATIVE,false,Difficulty.NORMAL,true,new GameRules(),WorldDataConfiguration.DEFAULT);
                mc.createWorldOpenFlows().createFreshLevel("qa_kitchen",settings,new WorldOptions(77L,false,false),access->access.registryOrThrow(Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions());
            }
            return;
        }
        if(mc.player==null||mc.level==null||mc.getSingleplayerServer()==null)return;
        if(!built){if(++tick<200)return;built=true;sx=(int)Math.floor(mc.player.getX());sy=(int)Math.floor(mc.player.getY());sz=(int)Math.floor(mc.player.getZ());script();}
        if(holding)return;
        if(wait>0){wait--;return;}
        var head=STEPS.peek();
        if(head!=null&&head.length>3){boolean ok;try{ok=((java.util.function.BooleanSupplier)head[3]).getAsBoolean();}catch(Throwable t){ok=false;}int[] rem=(int[])head[4];if(!ok&&rem[0]-->0){wait=3;return;}if(!ok)com.mojang.logging.LogUtils.getLogger().info("QA_TIMEOUT {}",head[2]);}
        var step=STEPS.poll();
        if(step==null){com.mojang.logging.LogUtils.getLogger().info("QA_DONE pass={} fail={}",pass,fail);mc.stop();return;}
        try{((Runnable)step[1]).run();}catch(Throwable t){fail++;com.mojang.logging.LogUtils.getLogger().error("QA_STEP_ERROR {}",step[2],t);}
        wait=(int)step[0];
    }

    static boolean oldDoubles(boolean stew,int mk,int first,int second,boolean legendary){return stew&&mk>=4&&first>=2&&(!legendary||second>=2);}

    static OldPreview oldCompose(net.minecraft.world.Container ingredients,boolean stew,int mk,Set<MealRules.Effect> selected){
        var foods=new LinkedHashMap<net.minecraft.world.item.Item,EnumMap<MealRules.Effect,Integer>>();
        var groups=new HashMap<String,Set<net.minecraft.world.item.Item>>();boolean staple=false;
        for(int i=0;i<6;i++){
            var stack=ingredients.getItem(i);if(stack.isEmpty())continue;
            var matches=IngredientTraits.traits.stream().filter(t->t.ingredient().test(stack)).toList();
            if(matches.isEmpty())return new OldPreview(null,foods.size(),staple,"ingredient");
            var scores=foods.computeIfAbsent(stack.getItem(),item->new EnumMap<>(MealRules.Effect.class));
            groups.computeIfAbsent(matches.get(0).family(),k->new HashSet<>()).add(stack.getItem());
            for(var trait:matches){staple|=trait.staple();trait.effects().forEach((e,w)->{if(!e.gun())scores.merge(e,w,Math::max);});}
        }
        int limit=stew?3:2;var bonuses=new ArrayList<MealData.Bonus>();var used=new HashSet<net.minecraft.world.item.Item>();var reserved=new HashSet<String>();
        if(selected!=null){
            if(selected.isEmpty()||selected.size()>limit)return new OldPreview(null,foods.size(),staple,"selection");
            for(var mix:MealRules.MIXES)if(selected.contains(mix.effect())){
                var first=groups.get(mix.first());var second=groups.get(mix.second());
                if(first==null||second==null||!reserved.add(mix.first())||!reserved.add(mix.second()))return new OldPreview(null,foods.size(),staple,"selection");
                used.addAll(first);used.addAll(second);
                bonuses.add(new MealData.Bonus(mix.effect(),1,oldDoubles(stew,mk,first.size(),second.size(),true)));
            }
        }
        var scores=new EnumMap<MealRules.Effect,Integer>(MealRules.Effect.class);
        foods.forEach((item,values)->{if(!used.contains(item))values.forEach((e,w)->scores.merge(e,w,Integer::sum));});
        var ordinary=selected==null?MealRules.select(scores,stew):selected.stream().filter(e->!e.gun()).sorted().toList();
        for(var e:ordinary){
            if(!scores.containsKey(e))return new OldPreview(null,foods.size(),staple,"selection");
            boolean doubled=groups.values().stream().anyMatch(items->items.stream().filter(item->!used.contains(item)&&foods.get(item).containsKey(e)).count()>=2);
            bonuses.add(new MealData.Bonus(e,1,oldDoubles(stew,mk,doubled?2:1,0,false)));
        }
        int occupied=foods.size();
        String problem=occupied<2?"ingredients":!stew&&occupied>3?"sandwich_types":bonuses.isEmpty()?"effects":"";
        return new OldPreview(problem.isEmpty()?new MealData(stew,bonuses.stream().limit(limit).toList(),MealRules.FIELD_TICKS):null,occupied,staple,problem);
    }
    static void equivalence(){
        var foods=new ArrayList<>(IngredientTraits.catalogue().items().values());foods.add(Items.STONE);foods.add(Items.DIAMOND);
        var random=new Random(42);var values=MealRules.Effect.values();int trials=6000,same=0,mapped=0,mismatched=0;var shown=0;
        for(int t=0;t<trials;t++){
            var c=new net.minecraft.world.SimpleContainer(8);int n=random.nextInt(7);
            for(int i=0;i<n;i++){var item=foods.get(random.nextInt(foods.size()));c.setItem(random.nextInt(6),new ItemStack(item,1+random.nextInt(5)));}
            boolean stew=true;int mk=4;Set<MealRules.Effect> selected=null; // 0.0.11: only a Mk IV stew still follows the v6 rules
            if(random.nextInt(3)>0){selected=EnumSet.noneOf(MealRules.Effect.class);int k=random.nextInt(4);while(selected.size()<k)selected.add(values[random.nextInt(values.length)]);}
            var before=oldCompose(c,stew,mk,selected);var after=IngredientTraits.compose(c,stew,mk,selected);
            boolean ok;
            if(before.meal()!=null||after.meal()!=null)ok=before.meal()!=null&&after.meal()!=null&&before.meal().equals(after.meal())&&before.occupied()==after.occupied();
            else if(before.problem().equals(after.problem()))ok=true;
            else if(before.problem().equals("selection")&&after.problem().equals("missing")){ok=true;mapped++;}
            else ok=false;
            if(ok)same++;else{mismatched++;if(shown++<5)com.mojang.logging.LogUtils.getLogger().info("QA_NOTE differs stew={} mk={} selected={} old={}/{} new={}/{}",stew,mk,selected,before.problem(),before.meal(),after.problem(),after.meal());}
        }
        check("compose is equivalent to the v6 implementation on "+trials+" random Mk IV stew tables of real foods ("+mapped+" rejections became 'missing')",mismatched==0,"same="+same+" differ="+mismatched);
    }
