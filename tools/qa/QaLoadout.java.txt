    // The walk as the client sees it: distance from the synced path and jumps between ticks
    static double maxDev,maxStep,startLag;static int samples,jumps;static Vec3 last;static boolean watchWalk;static int cinematicTicks,detachedTicks;
    @SubscribeEvent public static void walk(TickEvent.ClientTickEvent e){
        if(e.phase!=TickEvent.Phase.END||mc().player==null||mc().level==null)return;
        var be=LoadoutClient.held();
        if(be==null||!watchWalk){last=null;return;}
        float t=be.elapsed(mc().level.getGameTime(),0);if(!be.timeline().holding(t)){last=null;return;}
        var p=mc().player.position();double best=Double.MAX_VALUE;
        for(int dt=-1;dt<=1;dt++)best=Math.min(best,p.distanceTo(LoadoutClient.at(be,t+dt)));
        if(t>=10)maxDev=Math.max(maxDev,best);else startLag=Math.max(startLag,best);samples++;
        if(last!=null){double d=p.distanceTo(last);maxStep=Math.max(maxStep,d);if(d>.5){jumps++;note("JUMP at t={} from {} to {} (synced {})",t,last,p,LoadoutClient.at(be,t));}}
        last=p;
        if(LoadoutCinematic.running())cinematicTicks++;
        if(mc().gameRenderer.getMainCamera().isDetached())detachedTicks++;
    }
    static void resetWalk(){startLag=0;maxDev=0;maxStep=0;samples=0;jumps=0;last=null;cinematicTicks=0;detachedTicks=0;hudDrawn=0;hudHidden=0;}

    // The driver
    @SubscribeEvent public static void onTick(TickEvent.ClientTickEvent e){
        if(e.phase!=TickEvent.Phase.END)return;var mc=mc();
        if(!created){
            if(mc.screen instanceof TitleScreen&&mc.getOverlay()==null&&++tick>60){
                created=true;var settings=new LevelSettings("qa_loadout",GameType.CREATIVE,false,Difficulty.NORMAL,true,new GameRules(),WorldDataConfiguration.DEFAULT);
                mc.createWorldOpenFlows().createFreshLevel("qa_loadout",settings,new WorldOptions(42L,false,false),access->access.registryOrThrow(Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions());
            }
            return;
        }
        if(mc.player==null||mc.level==null||mc.getSingleplayerServer()==null)return;
        if(!built){if(++tick<200)return;built=true;sx=(int)Math.floor(mc.player.getX());sy=(int)Math.floor(mc.player.getY());sz=(int)Math.floor(mc.player.getZ());places();script();}
        if(holding)return;
        if(wait>0){wait--;return;}
        var head=STEPS.peek();
        if(head!=null&&head.length>3){boolean ok;try{ok=((java.util.function.BooleanSupplier)head[3]).getAsBoolean();}catch(Throwable t){ok=false;}int[] rem=(int[])head[4];if(!ok&&rem[0]-->0){wait=0;return;}if(!ok)log("QA_TIMEOUT {}",head[2]);}
        var step=STEPS.poll();
        if(step==null){log("QA_DONE pass={} fail={}",pass,fail);mc.stop();return;}
        log("QA_STEP {}",step[2]);
        try{((Runnable)step[1]).run();}catch(Throwable t){fail++;com.mojang.logging.LogUtils.getLogger().error("QA_STEP_ERROR {}",step[2],t);}
        wait=(int)step[0];
    }

    /** A print stopped before tick 40 by how: nothing consumed, the gear as it was, the camera back. */
    static void stopCase(String how,Runnable stop,boolean respawn){
        List<ItemStack> before=new ArrayList<>();int[] coins={0};
        step(5,()->{readyNow();face();dressPlain();clearMessages();},how+": in front");
        if(how.equals("logout")){
            // A player who is not in the player list (a fake player) is treated as gone on the next tick
            step(5,()->{var rv=rival();String r=srvGet(()->{var m=LoadoutStation.machine(sl(),S);int c=m.coins[0].getCount();rv.moveTo(S.getX()+.5,sy,S.getZ()-3.5,0,0);rv.getInventory().clearContent();
                LoadoutData.get(sl()).players.setReadyAt(rv.getUUID(),0);LoadoutPrint.request(rv,m,0);boolean started=LoadoutPrint.of(rv.getUUID())!=null;return c+" "+started;});coins[0]=Integer.parseInt(r.split(" ")[0]);note("logout case: {}",r);},"a player not in the player list starts a print");
            step(5,()->check("logout before tick 40: the print stops, the coin stays",srvGet(()->LoadoutPrint.of(rival().getUUID())==null&&LoadoutPrint.at(LoadoutStation.machine(sl(),S).id)==null)&&stock(0)==coins[0],""+stock(0)),"logout");
            return;
        }
        step(2,()->{coins[0]=stock(0);before.clear();before.addAll(worn());note("{} inventory before: {}",how,inventory());click(S.offset(0,-1,-1),Direction.NORTH);},how+": start a print");
        until(how+": tick 20",100,()->clientElapsed()>=20,()->{check(how+": the camera shot is on before the stop",LoadoutCinematic.running(),"");stop.run();});
        step(20,()->{
            if(respawn&&mc().player.isDeadOrDying())mc().player.respawn();
        },how+": respawn if dead");
        int[] presses={0};
        until(how+": alive again",300,()->{var p=mc().player;if(p==null)return false;if(p.isDeadOrDying()){if(presses[0]++%20==0)p.respawn();return false;}return true;},()->{if(mc().screen instanceof net.minecraft.client.gui.screens.DeathScreen)mc().setScreen(null);});
        until(how+": the client is ready",400,()->mc().player!=null&&mc().level!=null&&mc().screen==null&&!mc().isPaused()&&!mc().player.isDeadOrDying()&&srvGet(()->sp().level().dimension())==mc().level.dimension(),()->note("{} inventory after: {}",how,inventory()));
        step(10,()->{},how+": a few frames in the new place");
        step(20,()->{
            check(how+" before tick 40: the print stops, the coin stays, the gear is unchanged",srvGet(()->LoadoutPrint.of(sp().getUUID())==null)&&stock(0)==coins[0]&&same(worn(),before),stock(0)+" "+diff(worn(),before));
            check(how+": the camera is back in first person, HUD on",firstPerson()&&!mc().options.hideGui,cameraState());
            check(how+": doors close (or the station is idle)",srvGet(()->{var be=(LoadoutStation.StationEntity)sl().getBlockEntity(S);return be.occupant==null;}),"");
            shot("lo3x_stop_"+how);
        },how+": after");
        step(10,()->srv(()->{if(sp().level()!=sl())sp().teleportTo(sl(),S.getX()+.5,sy,S.getZ()-3.5,0,10);}),how+": back home");
        settled(how);
    }
