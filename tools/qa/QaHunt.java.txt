 * and the alarm's rhythm (a start every 22 client ticks). 0.0.22: a supply and a return flare answer with the owner's own radio lines.

    }

    /**
     * Server tick time while {@link #timing} is on (ServerTickEvent START to END: the whole tick, every level), per stage of the arrival:
     * {@link #label} when the script sets one, else read from the mission (before contact, spawning, after spawning).
     */
    static long tickStart;static volatile boolean timing;static volatile String label;
    static final Map<String,long[]> STAGES=new java.util.concurrent.ConcurrentHashMap<>();   // stage -> ticks, slowest ns, total ns
    @SubscribeEvent public static void serverTick(TickEvent.ServerTickEvent e){
        if(e.phase==TickEvent.Phase.START){tickStart=System.nanoTime();return;}
        if(!timing)return;long ns=System.nanoTime()-tickStart;
        String stage=label;if(stage==null){var m=HuntData.get(sl()).board.mission;stage=m==null?"no mission":!m.engaged()?"walking in, before contact":m.pending()>0?"spawning the warband":"after the warband is out";}
        STAGES.computeIfAbsent(stage,k->new long[3]);var v=STAGES.get(stage);v[0]++;v[1]=Math.max(v[1],ns);v[2]+=ns;
        if(ns>30_000_000L)log("QA_NOTE slow server tick: {} ms at game time {} ({})",ns/1_000_000,sl().getGameTime(),stage);
    }
    static long slowestMs(String stage){var v=STAGES.get(stage);return v==null?-1:v[1]/1_000_000;}
    static double meanMs(String stage){var v=STAGES.get(stage);return v==null||v[0]==0?-1:v[2]/(double)v[0]/1_000_000;}

    }
    @SubscribeEvent public static void chat(net.minecraftforge.client.event.ClientChatReceivedEvent e){String t=e.getMessage().getString();CHAT.add(t);log("QA_CHAT {}",t);}

    @SubscribeEvent public static void onTick(TickEvent.ClientTickEvent e){
        if(e.phase!=TickEvent.Phase.END)return;var mc=mc();ctick++;
        if(!created){
            if(mc.screen instanceof TitleScreen&&mc.getOverlay()==null&&++tick>60){
                created=true;
                var settings=new LevelSettings("qa_hunt",GameType.CREATIVE,false,Difficulty.NORMAL,true,new GameRules(),WorldDataConfiguration.DEFAULT);
                mc.createWorldOpenFlows().createFreshLevel("qa_hunt",settings,new WorldOptions(20201010L,true,false),access->access.registryOrThrow(Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.NORMAL).value().createWorldDimensions());
            }
            return;
        }
        if(mc.player==null||mc.level==null||mc.getSingleplayerServer()==null)return;
        if(!started){
            if(++tick<300)return;started=true;
            cmd("gamerule doMobSpawning false");cmd("gamerule doDaylightCycle false");cmd("weather clear");cmd("time set 6000");
            srv(()->{MinecraftForge.EVENT_BUS.register(new HuntGameTests.Runner());HuntGameTests.Runner.start(sl());});
            return;
        }
        if(!testsDone){
            if(HuntGameTests.Runner.lastPass<0)return;
            testsDone=true;check("hunt GameTests",HuntGameTests.Runner.lastFail==0,"pass="+HuntGameTests.Runner.lastPass+" fail="+HuntGameTests.Runner.lastFail);
            script();return;
        }
        if(wait>0){wait--;return;}
        var head=STEPS.peek();
        if(head!=null&&head.length>3){boolean ok;try{ok=((java.util.function.BooleanSupplier)head[3]).getAsBoolean();}catch(Throwable t){ok=false;}int[] rem=(int[])head[4];if(!ok&&rem[0]-->0){wait=3;return;}if(!ok){fail++;log("QA_CHECK FAIL timeout {}","" + head[2]);}}
        var step=STEPS.poll();
        if(step==null){log("QA_DONE pass={} fail={}",pass,fail);mc.stop();return;}
        try{((Runnable)step[1]).run();}catch(Throwable t){fail++;com.mojang.logging.LogUtils.getLogger().error("QA_STEP_ERROR {}",step[2],t);}
        wait=(int)step[0];
    }
