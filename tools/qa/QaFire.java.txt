        step(5,()->check("death clears Ablaze",srvGet(()->!dry.isAlive()&&!dry.getPersistentData().contains(FireMarks.ABLAZE)),""),"death");

        // ==== 2.
        step(5,()->srv(()->{
            SPOTTED.clear();var r=new Random(5);
            for(int i=0;i<12;i++)SPOTTED.add(mob(EntityType.HUSK,ARROWS(),r.nextDouble()*6-3,r.nextDouble()*6-3,300,false));

            KNOCKS.clear();}),"15 hostile mobs in the box (more than the old cap of 10), one a raid boss");
        step(5,()->look(ARROWS().offset(0,0,-11),3,ARROWS().getX()+.5,ARROWS().getY()+1,ARROWS().getZ()+.5),"watch");

            check("each is Glowing for good (an endless effect)",SPOTTED.stream().allMatch(m->endless(m,MobEffects.GLOWING,0)),"");
            boolean firstHits=true;for(var m:SPOTTED){var h=hurts(m,"minecraft:arrow");if(h.size()!=1||h.get(0).pre()!=4f||h.get(0).post()!=4f)firstHits=false;}
            check("each took one arrow hit of 4 (the mark comes after the hit)",firstHits,show(hurts(SPOTTED.get(0),"minecraft:arrow")));
            boolean noKnock=true;int tried=0;synchronized(KNOCKS){for(var k:KNOCKS){int id=(int)k[0];if(SPOTTED.stream().anyMatch(m->m.getId()==id)){tried++;if(!(boolean)k[1])noKnock=false;}}}
            check("no knockback from the hit (Forge's knockback event is cancelled for it)",noKnock&&tried>=15,tried+" knockbacks, all cancelled: "+noKnock);
            check("the falling arrows hit nothing",SPOTTED.stream().allMatch(m->hurts(m,null).stream().noneMatch(Hurt::byArrow)),"");},"first volley");
        step(10,()->{check("the client sees them glow",SPOTTED.stream().allMatch(m->client(m)!=null&&client(m).isCurrentlyGlowing()),"");
            check("arrows fall (for show)",srvGet(()->!sl().getEntitiesOfClass(net.minecraft.world.entity.projectile.Arrow.class,new AABB(ARROWS()).inflate(12)).isEmpty()),"");shot("f06_spotter_marked");},"glow");
        step(5,()->srv(()->{late=mob(EntityType.HUSK,ARROWS(),1,1,300,false);}),"a mob arrives late");
        until("the next volley",200,()->marked(late),()->{});
        step(3,()->{
            check("a mob that arrived late is marked by the next volley",marked(late)&&endless(late,MobEffects.GLOWING,0),"");
            var second=hurts(SPOTTED.get(0),"minecraft:arrow");
            check("a marked mob takes 30 percent more from the next hit (4 -> 5.2)",second.size()>=2&&Math.abs(second.get(1).post()-5.2f)<1e-3,show(second));
            var b=hurts(boss,"minecraft:arrow");
            check("a raid boss takes 15 percent more (4 -> 4.6)",b.size()>=2&&Math.abs(b.get(1).post()-4.6f)<1e-3,show(b));},"late arrival");
        until("the volleys end",900,()->!flareAt(ARROWS()),()->{});
        until("the arrows vanish",200,()->srvGet(()->sl().getEntitiesOfClass(net.minecraft.world.entity.projectile.Arrow.class,new AABB(ARROWS()).inflate(12)).isEmpty()),()->{});
        step(2,()->{
            check("the show arrows are gone soon after they land",srvGet(()->sl().getEntitiesOfClass(net.minecraft.world.entity.projectile.Arrow.class,new AABB(ARROWS()).inflate(12)).isEmpty()),"");
            check("six volleys hit each mob six times",hurts(SPOTTED.get(0),"minecraft:arrow").size()==6,""+hurts(SPOTTED.get(0),"minecraft:arrow").size());
            check("marks do not stack: every later hit is 5.2",hurts(SPOTTED.get(0),"minecraft:arrow").stream().skip(1).allMatch(h->Math.abs(h.post()-5.2f)<1e-3),show(hurts(SPOTTED.get(0),"minecraft:arrow")));},"after the volleys");

        // Marked: every kind of damage, a real TaCZ bullet among them; what ignores invulnerability is left alone
        step(5,()->srv(()->{
            vind=mob(EntityType.VINDICATOR,PROBES(),0,0,500,false);probe=mob(EntityType.HUSK,PROBES(),3,0,500,false);bypass=mob(EntityType.HUSK,PROBES(),-3,0,500,false);
            control=mob(EntityType.VINDICATOR,PROBES(),0,4,500,false);taczMarked=mob(EntityType.VINDICATOR,PROBES(),-2,8,500,false);taczPlain=mob(EntityType.VINDICATOR,PROBES(),2,8,500,false);
            for(var m:List.of(vind,probe,bypass,taczMarked))FireMarks.mark(m,sp());}),"marked mobs to probe, an unmarked control");
        step(15,()->srv(()->{var s=sl().damageSources();vind.hurt(s.explosion(null,null),10f);control.hurt(s.explosion(null,null),10f);}),"an explosion");
        step(15,()->srv(()->{var s=sl().damageSources();vind.hurt(s.inFire(),3f);control.hurt(s.inFire(),3f);}),"fire");
        step(15,()->srv(()->{var s=sl().damageSources();probe.hurt(s.mobAttack(control),6f);}),"a melee hit");
        step(80,()->srv(()->{vind.addEffect(new MobEffectInstance(MobEffects.POISON,200,1));control.addEffect(new MobEffectInstance(MobEffects.POISON,200,1));}),"poison, four seconds");
        step(80,()->srv(()->{vind.removeEffect(MobEffects.POISON);control.removeEffect(MobEffects.POISON);vind.addEffect(new MobEffectInstance(MobEffects.WITHER,200,1));control.addEffect(new MobEffectInstance(MobEffects.WITHER,200,1));}),"wither, four seconds");
        step(80,()->srv(()->{vind.removeEffect(MobEffects.WITHER);control.removeEffect(MobEffects.WITHER);bypass.kill();}),"a kill that ignores invulnerability");
        step(5,()->{
            check("Marked: an explosion counts 30 percent more",ratio(hurts(vind,"minecraft:explosion"),1.3),show(hurts(vind,"minecraft:explosion")));
            check("Marked: fire counts 30 percent more",ratio(hurts(vind,"minecraft:in_fire"),1.3),show(hurts(vind,"minecraft:in_fire")));
            check("Marked: a melee hit counts 30 percent more",ratio(hurts(probe,"minecraft:mob_attack"),1.3),show(hurts(probe,"minecraft:mob_attack")));
            check("Marked: every poison tick counts 30 percent more",hurts(vind,"minecraft:magic").size()>=3&&ratio(hurts(vind,"minecraft:magic"),1.3),show(hurts(vind,"minecraft:magic")));
            check("Marked: every wither tick counts 30 percent more",hurts(vind,"minecraft:wither").size()>=2&&ratio(hurts(vind,"minecraft:wither"),1.3),show(hurts(vind,"minecraft:wither")));
            check("damage that ignores invulnerability (a kill) is left alone",ratio(hurts(bypass,"minecraft:generic_kill"),1.0),show(hurts(bypass,"minecraft:generic_kill")));
            check("an unmarked mob takes what it always took (explosion, fire, poison, wither)",ratio(hurts(control,null),1.0)&&!hurts(control,"minecraft:magic").isEmpty()&&!hurts(control,"minecraft:wither").isEmpty(),show(hurts(control,null)));},"probes");
        step(5,()->{look(PROBES().offset(0,0,4),0,taczMarked.getX(),taczMarked.getY()+1,taczMarked.getZ());gun();},"a TaCZ AK-47 in hand, 4 blocks from the targets");
        step(40,()->{},"draw the gun");
        for(int k=0;k<6;k++){
            step(20,()->{if(hurts(taczMarked,null).stream().noneMatch(h->h.type().startsWith("tacz:")))log("QA_NOTE shoot marked: {}",shoot(taczMarked));},"shoot the marked mob");
        }
        for(int k=0;k<6;k++){
            step(20,()->{if(hurts(taczPlain,null).stream().noneMatch(h->h.type().startsWith("tacz:")))log("QA_NOTE shoot unmarked: {}",shoot(taczPlain));},"shoot the unmarked mob");
        }
        step(10,()->{
            var bullets=hurts(taczMarked,null).stream().filter(h->h.type().startsWith("tacz:")).toList();var plain=hurts(taczPlain,null).stream().filter(h->h.type().startsWith("tacz:")).toList();
            check("a real TaCZ bullet hit the marked mob",!bullets.isEmpty(),show(hurts(taczMarked,null)));
            check("Marked: the TaCZ bullet counts 30 percent more",ratio(bullets,1.3),show(bullets));
            check("an unmarked mob takes the bullet as it is",!plain.isEmpty()&&ratio(plain,1.0),show(plain));
            srv(()->sp().setItemInHand(InteractionHand.MAIN_HAND,ItemStack.EMPTY));},"TaCZ");

        // ==== 3.
        step(5,()->srv(()->{CURSED.clear();for(int i=0;i<3;i++)CURSED.add(mob(EntityType.VINDICATOR,CURSE(),i*2-2,0,400,false));CURSED.add(mob(EntityType.HUSK,CURSE(),0,2,400,false));}),"three vindicators and a husk");
        step(5,()->look(CURSE().offset(0,0,-9),2,CURSE().getX()+.5,CURSE().getY()+1,CURSE().getZ()+.5),"watch");
        step(5,()->flare(CannonUpgrades.FireType.CURSE,CURSE(),3),"a Curse of Debilitation flare, Damage level 3");
        until("the first shell",600,()->srvGet(()->CURSED.get(0).hasEffect(MobEffects.WITHER)),()->{});
        step(5,()->{
            boolean all=true;for(int i=0;i<3;i++){var m=CURSED.get(i);all&=endless(m,MobEffects.WITHER,1)&&endless(m,MobEffects.POISON,1)&&endless(m,MobEffects.WEAKNESS,1)&&endless(m,MobEffects.MOVEMENT_SLOWDOWN,2)&&endless(m,MobEffects.BLINDNESS,0);}
            check("one shell: Wither II, Poison II, Weakness II, Slowness III and Blindness, all endless (Damage level 3 changes nothing)",all,srvGet(()->CURSED.get(0).getActiveEffects().toString()));
            check("no Glowing any more",CURSED.stream().noneMatch(m->srvGet(()->m.hasEffect(MobEffects.GLOWING))),"");
            var h=CURSED.get(3);check("an undead husk gets them too, except Poison (vanilla: undead are immune)",endless(h,MobEffects.WITHER,1)&&endless(h,MobEffects.BLINDNESS,0)&&!srvGet(()->h.hasEffect(MobEffects.POISON)),"");
            shot("f07_curse");},"cursed");
        until("the curse's shells end",900,()->!flareAt(CURSE()),()->{});
        step(200,()->{},"ten seconds later");
        step(2,()->check("the curses are still there, still endless",CURSED.stream().limit(3).allMatch(m->endless(m,MobEffects.WITHER,1)&&endless(m,MobEffects.WEAKNESS,1)),""),"still cursed");

        // ==== 4.
        Mob[] raid=new Mob[5];
        step(5,()->srv(()->{
            raid[0]=mob(EntityType.HUSK,RAID(),-12,0,200,false);
            raid[1]=mob(EntityType.HUSK,RAID(),0,0,200,false);raid[2]=mob(EntityType.VINDICATOR,RAID(),0,10,200,false);raid[3]=mob(EntityType.HUSK,RAID(),8,-5,200,false);raid[4]=mob(EntityType.HUSK,RAID(),20,0,200,false);
            for(var m:raid)m.getPersistentData().putString("eternalRole","heavy");
            var d=CampaignData.get(sl());d.phase="raid";d.raidType="normal";d.hardRaid=false;d.raidTier=10;d.introRaid=false;d.introCompleted=true;d.wave=1;d.waveTicks=0;d.raidTicks=0;d.raidsStarted=7;
            d.spawnRemaining=0;d.spawnCooldown=1_000_000;d.raiders.clear();for(var m:raid)d.raiders.add(m.getUUID());RaidAdaptation.begin(d);d.setDirty();}),"a raid: three husks and a vindicator are its raiders");
        step(5,()->{flare(CannonUpgrades.FireType.NAPALM,RAID().offset(-12,0,0),0);flare(CannonUpgrades.FireType.ARROW,RAID(),0);flare(CannonUpgrades.FireType.CURSE,RAID().offset(0,0,10),0);},"a napalm, Spotter Arrows and a curse on the raiders");

        step(2,()->{
            var learned=srvGet(()->new TreeMap<>(CampaignData.get(sl()).trapFirstWave).toString());
            check("in the raid the napalm's raider was hurt by the carpet and by Ablaze",!hurts(raid[0],"minecraft:in_fire").isEmpty()&&!hurts(raid[0],"minecraft:on_fire").isEmpty(),show(hurts(raid[0],null)));
            check("in the raid the Spotter's raider took its hits",!hurts(raid[1],"minecraft:arrow").isEmpty(),show(hurts(raid[1],null)));
            check("in the raid the cursed raider took poison and wither ticks",!hurts(raid[2],"minecraft:magic").isEmpty()&&!hurts(raid[2],"minecraft:wither").isEmpty(),show(hurts(raid[2],null)));

        // ==== 5.
        step(2,()->mc().options.hideGui=false,"the HUD back for the screens");
        step(5,()->{var g=new BeaconClient.GuideScreen(null);mc().setScreen(g);},"open the Field Guide");

        for(Object[] c:new Object[][]{{1,"f10_menu_spotter",new String[]{"Spotter Arrows","glows"},"Arrow Cluster"},{5,"f11_menu_curse",new String[]{"Curse of Debilitation","dies"},"Glowing"},{8,"f12_menu_napalm",new String[]{"Napalm Carpet","Ablaze"},"tipped"}}){
            step(5,()->{var tag=new net.minecraft.nbt.CompoundTag();tag.putInt("fire",(int)c[0]);tag.putIntArray("up",new int[CannonUpgrades.Upgrade.values().length]);tag.putString("owner","QA");tag.putBoolean("creative",true);
                mc().setScreen(new CannonScreen(BlockPos.ZERO,tag));},"the cannon menu, type "+c[0]);
            step(15,()->{

                check("the cannon menu (open) shows the card for "+((String[])c[2])[0]+", which says "+String.join(", ",(String[])c[2]),all,card);
                check("and no longer says '"+c[3]+"' (and no placeholder is left)",!card.contains((String)c[3])&&!card.matches("(?s).*%(\\d\\$)?s.*"),card);
                shot((String)c[1]);},"card");
        }
        step(3,()->mc().setScreen(null),"close");
    }
