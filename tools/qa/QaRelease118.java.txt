            String s="client scale "+Elites.SCALES.get(big)+", kinds "+Elites.KINDS+", raiders "+RaidTypes.RAIDERS+", hitbox "+(client==null?"none":String.format(Locale.ROOT,"%.2fx%.2f",client.getBbWidth(),client.getBbHeight()));note("CLIENT {}",s);
            check("the client knows the giant's size and gives it the grown hitbox",Elites.SCALES.getOrDefault(big,0f)==5f&&client!=null&&Math.abs(client.getBbWidth()-srvGet(()->subject("knight5").getBbWidth()))<.01,s);
            boolean kinds=true;for(String k:List.of("knight5","kobo","zombie","hound"))kinds&=Elites.KINDS.containsKey(clientId(k));
            check("the client knows each elite's kind (for its colour)",kinds&&Elites.kindOf(Elites.KINDS.get(clientId("kobo")))==Elites.Kind.HEALER&&Elites.color(Elites.KINDS.get(big))==0xFF3B30,s);
            check("the client marks every raid mob, the GeckoLib ones too",RaidTypes.RAIDERS.contains(big)&&RaidTypes.RAIDERS.contains(plain)&&RaidTypes.RAIDERS.contains(clientId("hound")),s);},"client side");
        step(5,()->{look(ARENA.getX()+.5,sy+5,ARENA.getZ()-38.5,ARENA.getX()+.5,sy+4,ARENA.getZ()-4.5);AURAS[0]=BARS[0]=MARKS[0]=0;counted=0;},"look at them from 34 blocks, at night");
        for(int i=0;i<20;i++)step(1,QaRelease118::sample,"count what the overlay draws");
        step(1,()->{note("OVERLAY auras {} bars {} marks {} over {} ticks",AURAS[0],BARS[0],MARKS[0],counted);
            check("an aura and a health bar for each of the four elites, a mark for each of the five raid mobs",AURAS[0]==4&&BARS[0]==4&&MARKS[0]==5,"auras "+AURAS[0]+", bars "+BARS[0]+", marks "+MARKS[0]);},"overlay counts");
        step(10,()->shot("r118_night_elites"),"picture at night, no night vision");
        step(5,()->{int big=clientId("knight5"),plain=clientId("knight");note("TAGS scale {} inside {}",TAG_SCALE,TAG_INSIDE);
            Float sb=TAG_SCALE.get(big),sp=TAG_SCALE.get(plain);
            check("the GeckoLib giant's draw runs inside the size: its name tag is drawn in the five-times pose (then undone)",Boolean.TRUE.equals(TAG_INSIDE.get(big))&&sb!=null&&sp!=null&&Math.abs(sb/sp-5)<.05,"giant "+sb+" plain "+sp+" inside "+TAG_INSIDE.get(big));
            check("a mob that is not grown is drawn as it is",!Boolean.TRUE.equals(TAG_INSIDE.get(clientId("kobo")))&&!Boolean.TRUE.equals(TAG_INSIDE.get(plain)),String.valueOf(TAG_INSIDE));},"name tag pose");

        step(10,()->shot("r118_night_giant"),"picture of the giant at night");

        step(15,()->shot("r118_giant_hitbox"),"picture: the model fills its hitbox");
        step(5,()->{mc().getEntityRenderDispatcher().setRenderHitBoxes(false);srv(()->sp().removeEffect(MobEffects.NIGHT_VISION));},"hitboxes off");
        step(5,()->srv(()->{var d=cd();for(var id:new ArrayList<>(d.raiders)){var e=sl().getEntity(id);if(e!=null)e.discard();}d.raiders.clear();d.phase="preparation";d.spawnRemaining=0;d.setDirty();clearMobs();}),"end the staged raid");

        // 3. Another mod's boss keeps its own fight
        step(5,()->{String r=srvGet(()->{var b=new StringBuilder();int i=0;
            for(String id:List.of("minecraft:wither","cataclysm:netherite_monstrosity","cataclysm:ender_golem","minecraft:zombie","born_in_chaos_v1:door_knight")){
                var m=make(id);m.moveTo(B.getX()+10+i*8,sy,B.getZ()+20,0,0);m.setNoAi(true);sl().addFreshEntity(m);i++;

            return b.toString();});note("BOSSES {}",r);
            check("the Wither and Cataclysm's Netherite Monstrosity are never sent at the beacon",r.contains("minecraft:wither attached=false boss=true")&&r.contains("cataclysm:netherite_monstrosity attached=false boss=true"),r);

            check("hard raid level "+tier+": Cataclysm's "+(tier==3?"Ender Golem":"Kobolediator")+" comes awake (not a ravager)",tier==3?r.equals("spawned true cataclysm:ender_golem awake true"):r.equals("spawned true cataclysm:kobolediator awake true asleep false"),r);},"hard raid boss at level "+tier);

        // 5. The features that call other mods are all on
        step(5,()->{var rewards=srvGet(()->RaidRewards.completion(sl(),3,true));int tactical=0;var names=new ArrayList<String>();
            for(var s:rewards)if(s.hasTag()&&(s.getTag().contains("ConsumableId")||s.getTag().contains("ThrowableId"))){tactical++;names.add(s.getCount()+" "+s.getTag().getString(s.getTag().contains("ConsumableId")?"ConsumableId":"ThrowableId"));}
            check("a hard raid's cache at level 3 still holds LesRaisins' medical kits and the three grenades",tactical==4,String.valueOf(names));
            int prices=srvGet(()->AmmoCoins.damage(sl()).size());boolean bridge=MealGunBackend.verifyReloadBridge();Object explosive=staticField(MealGunCompat.class,"explosive");
            note("BACKENDS ammo prices {} reload bridge {} explosive field {}",prices,bridge,explosive);
            check("TaCZ Attributes' reload bridge, TaCZ's explosive-bullet field and its ammunition are found",bridge&&explosive!=null&&prices>10,"prices "+prices+", bridge "+bridge+", field "+explosive);
            var broken=BROKEN_NAMES.stream().filter(Compat::broken).toList();check("no feature switched itself off",broken.isEmpty(),String.valueOf(broken));},"the features that call other mods");
    }
