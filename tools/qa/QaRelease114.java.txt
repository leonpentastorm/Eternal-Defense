        wait=(int)step[0];
    }

        step(8,()->shot("r114_raid_special_tooltip"),"its tooltip");
        // The roomy layout (GUI scale 2)
        step(5,()->{hover(0,0);mc().options.guiScale().set(2);mc().resizeDisplay();},"GUI scale 2");
        step(10,()->{check("the roomy layout",screen().ph>=270,"ph "+screen().ph);quiet();recordFrame=true;},"modifiers, roomy");
        step(5,()->{check("roomy page buttons fit",buttonsCut().equals("[]"),buttonsCut());check("roomy modifiers page: texts fit, each modifier says what it does and pays",cut().equals("[]")&&drawnHas("30% more mobs every wave")&&drawnHas("+20% pay, +10% Ardent"),cut()+" "+drawn());shot("r114_raid_mods_roomy");press(labelled("Level & break"));},"Level & break, roomy");
        step(10,()->{quiet();recordFrame=true;},"redraw");
        step(5,()->{check("roomy level page: texts fit, the stats as chips",cut().equals("[]")&&drawnHas("Pays "),cut()+" "+drawn());shot("r114_raid_level_roomy");},"Level & break, roomy");
        step(5,()->{mc().options.guiScale().set(0);mc().resizeDisplay();},"back to auto");
        step(5,()->mc().setScreen(null),"close");

        // 6. The raid with its modifiers, to the end
        step(5,()->{CHAT.clear();KILLS.clear();ARDENT.clear();ELITE_DROPS.clear();srv(()->{sl().setDayTime(18000);var d=cd();d.raidsStarted=0;d.victories=0;d.setDirty();});},"midnight");
        step(5,()->srv(()->BeaconNetwork.open(sp(),"menu","")),"open the beacon");
        until("beacon screen",60,()->screen()!=null,()->{});
        step(5,()->press(labelled("Raid control")),"Raid control");
        step(5,()->press(screen().startRaid),"press Start raid");
        until("the raid starts",1200,()->srvGet(()->cd().phase).equals("raid"),()->{});
        until("the start's chat lines reach the client",100,()->said("Wave 1 / "),()->{});
        step(10,()->{var d=srvGet(()->List.copyOf(cd().activeMods)+" "+List.copyOf(cd().chosenMods)+" "+cd().raidType+" "+cd().raidTier+" "+cd().hardRaid+" "+cd().spawnRemaining+" "+cd().wavePlayers+" "+cd().waveVeteran);note("RAID {}",d);
            check("the raid runs with both modifiers; the choice is used up",d.startsWith("[reinforcements, elite_cadre] [] normal 1 false"),d);
            check("chat names them and what they pay",said("Raid modifiers: Reinforcements, Elite Cadre. Payout +35%, Ardent drops from raid mobs +25%."),saidLine("Raid modifiers"));
            String[] v=d.split(" ");int remaining=Integer.parseInt(v[v.length-3]),players=Integer.parseInt(v[v.length-2]),vet=Integer.parseInt(v[v.length-1]);
            int expected=RaidModifiers.attackers(Rules.waveEnemies(1,1,players,vet,false),"normal",List.of("reinforcements"));
            check("wave 1 sends the 15% larger count raised 30% by Reinforcements ("+expected+")",saidLine("Wave 1 / ").contains(expected+" attackers"),saidLine("Wave 1 / "));
            mc().setScreen(null);},"started");
        step(5,()->srv(()->BeaconNetwork.open(sp(),"menu","")),"the beacon during the raid");
        until("beacon screen",60,()->screen()!=null,()->{});
        step(5,()->press(labelled("Raid control")),"Raid control");
        step(5,()->press(screen().modsPage),"its modifiers");
        step(10,()->{var s=screen();check("during the raid the page shows this raid's modifiers, locked",s.modToggles.get(0).selected&&s.modToggles.get(1).selected&&s.modToggles.stream().noneMatch(b->b.active)&&s.specials.stream().noneMatch(b->b.active),"");quiet();recordFrame=true;},"locked");
        step(5,()->{check("and says what this raid pays",drawnHas("This raid: payout +35%, Ardent drops +25%"),drawn());shot("r114_raid_mods_running");mc().setScreen(null);},"running");
        // Only so many raiders live at once: the wave goes on as the plain ones fall
        int[] waitPoll={0};
        until("an elite shows up",2400,()->{if(said("an elite is here"))return true;if(++waitPoll[0]%60==0)srv(()->{for(var m:raiders())if(!Elites.elite(m))kill(m);});return false;},()->{});
        step(40,()->{String line=saidLine("an elite is here");note("ELITE LINE {}",line);
            check("chat announces the elite and the wave's extra Ardent",line.matches(".*Wave \\d: an elite is here, .*")&&line.endsWith("This wave pays 1 more Ardent Energy."),line);
            String r=srvGet(()->{var b=new StringBuilder();for(var m:raiders()){var k=Elites.kind(m);if(k!=null)b.append(k.id).append(" lvl ").append(m.getPersistentData().getInt(Elites.LEVEL)).append(String.format(Locale.ROOT," hp %.0f/%.0f ",m.getMaxHealth(),base(m))).append(m.getCustomName()==null?"no name":m.getCustomName().getString()).append(m.isCustomNameVisible()?" shown":" hidden").append("; ");}return b.toString();});
            note("ELITES {}",r);check("the elite carries its name over its head",r.contains(" shown")&&!r.contains("hidden"),r);
            check("eliteWaves counts the waves with an elite so far",srvGet(()->cd().eliteWaves)==srvGet(()->cd().wave),srvGet(()->cd().eliteWaves)+" of "+srvGet(()->cd().wave));
            var id=srvGet(()->raiders().stream().filter(Elites::elite).findFirst().map(m->new double[]{m.getX(),m.getY(),m.getZ()}).orElse(null));
            if(id!=null)look(id[0]+4,id[1],id[2]+4,id[0],id[1]+1,id[2]);},"the elite");
        step(20,()->shot("r114_raid_elite"),"picture");
        // Win: every raider falls to the player
        int[] poll={0};
        until("the raid is won",9000,()->{if(++poll[0]%10==0)srv(()->{for(var m:raiders())kill(m);});return !srvGet(()->cd().phase).equals("raid");},()->{});
        step(20,()->{var d=srvGet(()->cd().phase+" "+cd().victories+" "+cd().eliteWaves+" "+cd().raidTier+" "+ardentIn(cd().rewards)+" "+List.copyOf(cd().activeMods)+" "+cd().nextRaidType);note("AFTER {}",d);
            String[] v=d.split(" ");int waves=Rules.waves(1),eliteWaves=Integer.parseInt(v[2]);
            check("the raid is won",(v[0].equals("restore")||v[0].equals("preparation"))&&v[1].equals("1"),d);
            check("with Elite Cadre every wave had an elite ("+waves+" waves)",eliteWaves==waves,d);
            int raid=RaidModifiers.raised(Rules.raidArdent(1),35);
            check("the reward chest gets the raid's Ardent raised 35% ("+raid+") plus 1 a wave with an elite",Integer.parseInt(v[4])==raid+eliteWaves,d);
            check("chat says the cache is 35% bigger and pays the elite waves",said("Raid modifiers: the cache is 35% bigger.")&&said("Elite waves: +"+eliteWaves+" Ardent Energy in the reward chest."),saidLine("cache is")+" | "+saidLine("Elite waves"));
            check("after the raid: no modifiers, the next raid is ordinary, chat says where to choose",d.endsWith(" [] normal")&&said("Pick a special raid or modifiers in the beacon's Raid control tab."),d);
            note("DROPS kills {} ardent {} elites {}",KILLS,ARDENT,ELITE_DROPS);
            int marked=KILLS.getOrDefault("marked",0),got=ARDENT.getOrDefault("marked",0);
            check("raid mobs drop Ardent now and then (about 1 in 6 with +25%; here "+got+" from "+marked+")",marked>=20&&got>=1&&got<=marked/2,KILLS+" "+ARDENT);
            check("every elite of the raid dropped Ardent",ELITE_DROPS.stream().filter(s->s.contains(" gen 0: ")).noneMatch(s->s.endsWith(": 0")),String.valueOf(ELITE_DROPS));},"the payout");
    }
