/**
 * QA ONLY (never part of a release), 0.0.42: every gun of the loaded gun packs at the ammo table, with TaCZ Magazines. For each gun:
 * the first table Age at which a bullet that fits it is listed, the first at which a magazine is, and whether TaCZ Magazines gives the
 * gun a magazine family at all. Copy the owner's gun packs into run-client/tacz and the owner's taczmagazines-*.toml into run-client/config
 * before the run (setup-qa.sh with QA_MODS holding TaCZ Magazines). Results: QA_MAG lines, a QA_MAG_SUMMARY, QA_DONE.
 */
@Mod.EventBusSubscriber(modid=ArsenalBeacon.ID,value=Dist.CLIENT)
public final class QaMagazines {
    static final Deque<Object[]> STEPS=new ArrayDeque<>();
    static int wait,tick,pass,fail;static boolean created,built;static int sx,sy,sz;
    /** A step runs, then the next one waits {@code delay} ticks. */
    static void step(int delay,Runnable r,String name){STEPS.add(new Object[]{delay,r,name});}
    /** Waits {@code ticks}, then runs the check. */
    static void after(int ticks,Runnable check,String name){step(ticks,()->{},"wait: "+name);step(1,check,name);}
    static void until(String name,int timeout,java.util.function.BooleanSupplier cond,Runnable then){STEPS.add(new Object[]{0,then,name,cond,new int[]{timeout}});}
    static Minecraft mc(){return Minecraft.getInstance();}
    static net.minecraft.server.MinecraftServer server(){return mc().getSingleplayerServer();}
    static net.minecraft.server.level.ServerPlayer sp(){return server().getPlayerList().getPlayers().get(0);}
    static net.minecraft.server.level.ServerLevel sl(){return server().overworld();}
    static void srv(Runnable r){server().execute(r);}
    static <T> T srvGet(Supplier<T> s){return server().submit(s).join();}
    static void log(String f,Object... a){com.mojang.logging.LogUtils.getLogger().info(f,a);}
    static void check(String name,boolean ok,String detail){if(ok)pass++;else fail++;log("QA_CHECK {} {} {}",ok?"PASS":"FAIL",name,detail);}
    static int frames,due;static boolean holding;static String pendingShot="";
    static void shot(String name){mc().getToasts().clear();pendingShot=name;due=frames+6;holding=true;}
    @SubscribeEvent public static void frame(TickEvent.RenderTickEvent e){
        if(e.phase!=TickEvent.Phase.END)return;frames++;
        if(holding&&frames>=due){new File("shots").mkdirs();Screenshot.grab(new File("shots"),pendingShot+".png",mc().getMainRenderTarget(),m->{});log("QA_SHOT {}",pendingShot);holding=false;}
    }
    /** Every sound of the mod the client actually starts. */
    static final List<String> SOUNDS=Collections.synchronizedList(new ArrayList<>());
    @SubscribeEvent public static void sound(net.minecraftforge.client.event.sound.PlaySoundEvent e){
        var s=e.getOriginalSound();if(s==null||!s.getLocation().getNamespace().equals(ArsenalBeacon.ID))return;
        SOUNDS.add(s.getLocation().getPath());log("QA_SOUND {}",s.getLocation());
    }
    static long sounds(String prefix){synchronized(SOUNDS){return SOUNDS.stream().filter(s->s.startsWith(prefix)).count();}}
    /**
     * The container has no audio device, so the client's sound engine never starts and plays nothing (no PlaySoundEvent). What the
     * server sends is counted instead: every sound packet of the mod that reaches the client's connection (a handler in front of the
     * connection's own, in the integrated server's local channel the packets arrive as objects).
     */
    static boolean hooked;
    static void hookSounds(){
        if(hooked||mc().getConnection()==null)return;hooked=true;
        mc().getConnection().getConnection().channel().pipeline().addBefore("packet_handler","qa_sounds",new io.netty.channel.ChannelInboundHandlerAdapter(){
            @Override public void channelRead(io.netty.channel.ChannelHandlerContext ctx,Object msg)throws Exception{
                ResourceLocation id=null;
                if(msg instanceof net.minecraft.network.protocol.game.ClientboundSoundPacket p)id=p.getSound().value().getLocation();
                else if(msg instanceof net.minecraft.network.protocol.game.ClientboundSoundEntityPacket p)id=p.getSound().value().getLocation();
                if(id!=null&&id.getNamespace().equals(ArsenalBeacon.ID)){SOUNDS.add(id.getPath());log("QA_SOUND_PACKET {}",id);}
                super.channelRead(ctx,msg);
            }
        });
        log("QA_NOTE sound packets are counted at the client's connection");
    }

    // ---- one frame's texts (Ui records them) ------------------------------------------------------------------------------------------
    static boolean recordFrame;static List<Ui.Drawn> lastFrame=List.of();
    @SubscribeEvent public static void renderPre(net.minecraftforge.client.event.ScreenEvent.Render.Pre e){if(recordFrame)Ui.drawn=new ArrayList<>();}
    @SubscribeEvent public static void renderPost(net.minecraftforge.client.event.ScreenEvent.Render.Post e){if(recordFrame&&Ui.drawn!=null){lastFrame=Ui.drawn;Ui.drawn=null;recordFrame=false;}}
    static boolean drawnHas(String part){for(var d:lastFrame)if(d.text()!=null&&d.text().contains(part))return true;return false;}
    static String drawn(){return lastFrame.stream().map(Ui.Drawn::text).filter(Objects::nonNull).toList().toString();}

    // ---- the client's own actions ------------------------------------------------------------------------------------------------------
    static void tp(double x,double y,double z,float yaw,float pitch){srv(()->sp().teleportTo(sl(),x,y,z,yaw,pitch));mc().player.setYRot(yaw);mc().player.setXRot(pitch);}
    static void look(BlockPos from,double tx,double ty,double tz){
        double x=from.getX()+.5,y=from.getY(),z=from.getZ()+.5,dx=tx-x,dy=ty-(y+1.62),dz=tz-z;
        tp(x,y,z,(float)Math.toDegrees(Math.atan2(-dx,dz)),(float)-Math.toDegrees(Math.atan2(dy,Math.sqrt(dx*dx+dz*dz))));
    }
    static void hold(ItemStack s){srv(()->{sp().getInventory().setItem(0,s);sp().getInventory().selected=0;});mc().player.getInventory().selected=0;}
    static void click(BlockPos pos,Direction face){
        var hit=new BlockHitResult(Vec3.atCenterOf(pos).add(face.getStepX()*.5,face.getStepY()*.5,face.getStepZ()*.5),face,pos,false);
        mc().gameMode.useItemOn(mc().player,InteractionHand.MAIN_HAND,hit);
    }
    static void press(net.minecraft.client.gui.components.AbstractWidget w){var s=mc().screen;double x=w.getX()+w.getWidth()/2.0,y=w.getY()+w.getHeight()/2.0;s.mouseClicked(x,y,0);s.mouseReleased(x,y,0);}
    static net.minecraft.client.gui.components.AbstractWidget button(String label){
        if(mc().screen==null)return null;for(var c:mc().screen.children())if(c instanceof net.minecraft.client.gui.components.AbstractWidget b&&b.getMessage().getString().equals(label))return b;return null;
    }
    static void closeAll(){mc().setScreen(null);srv(()->sp().closeContainer());}
    static boolean loaded(BlockPos pos){return srvGet(()->sl().hasChunkAt(pos))&&mc().level!=null&&mc().level.hasChunkAt(pos);}

    // ---- the driver --------------------------------------------------------------------------------------------------------------------
    @SubscribeEvent public static void onTick(TickEvent.ClientTickEvent e){
        if(e.phase!=TickEvent.Phase.END)return;var mc=mc();
        if(!created){
            if(mc.screen instanceof TitleScreen&&mc.getOverlay()==null&&++tick>60){
                created=true;var settings=new LevelSettings("qa_r42",GameType.CREATIVE,false,Difficulty.PEACEFUL,true,new GameRules(),WorldDataConfiguration.DEFAULT);
                mc.createWorldOpenFlows().createFreshLevel("qa_r42",settings,new WorldOptions(42L,false,false),access->access.registryOrThrow(Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions());
            }
            return;
        }
        if(mc.player==null||mc.level==null||mc.getSingleplayerServer()==null)return;
        if(!built){if(++tick<200)return;built=true;hookSounds();sx=(int)Math.floor(mc.player.getX());sy=(int)Math.floor(mc.player.getY());sz=(int)Math.floor(mc.player.getZ());script();}
        if(holding)return;
        if(wait>0){wait--;return;}
        var head=STEPS.peek();
        if(head!=null&&head.length>3){boolean ok;try{ok=((java.util.function.BooleanSupplier)head[3]).getAsBoolean();}catch(Throwable t){ok=false;}int[] rem=(int[])head[4];if(!ok&&rem[0]-->0){wait=3;return;}if(!ok)log("QA_TIMEOUT {}",head[2]);}
        var step=STEPS.poll();
        if(step==null){log("QA_DONE pass={} fail={}",pass,fail);mc.stop();return;}
        log("QA_STEP {}",step[2]);
        try{((Runnable)step[1]).run();}catch(Throwable t){fail++;com.mojang.logging.LogUtils.getLogger().error("QA_STEP_ERROR {}",step[2],t);}
        wait=(int)step[0];
    }

    static ItemStack st(Item item,int n){return new ItemStack(item,n);}
    static volatile boolean scanned;
    static void script(){
        step(40,()->srv(()->{sp().setGameMode(GameType.SURVIVAL);}),"survival");
        step(5,()->srv(()->{
            var api=com.tacz.guns.api.TimelessAPI.getAllCommonGunIndex();var ids=new ArrayList<ResourceLocation>();for(var e:api)ids.add(e.getKey());ids.sort(Comparator.comparing(ResourceLocation::toString));
            int guns=0,inSystem=0,noMag=0,later=0,never=0,same=0,earlier=0,noBullet=0;var laterList=new ArrayList<String>();var neverList=new ArrayList<String>();
            Class<?> fam=null;try{fam=Class.forName("com.raiiiden.taczmagazines.magazine.MagazineFamilySystem");}catch(Exception x){}
            for(var id:ids){
                ItemStack gun;try{gun=MessHallV4GameTests.gun(sp(),id.toString());}catch(Exception x){continue;}
                if(!WeaponCompatibility.craftable(gun))continue;guns++;
                String family=null;try{family=(String)fam.getMethod("getFamilyForGun",ResourceLocation.class).invoke(null,id);}catch(Exception x){}
                if(family!=null)inSystem++;
                var all=WeaponPlatform.entries(sp(),"ammo","").stream().filter(e->WeaponPlatform.compatible(sp(),e)).toList();
                int bulletAge=99,magAge=99;int mags=0;
                for(var e:all){boolean mag=MagazineBridge.isMagazine(e.output());if(mag)mags++;
                    for(int age=1;age<=5;age++)if(CreateUnlocks.unlocked(sp(),e,age)){if(mag)magAge=Math.min(magAge,age);else bulletAge=Math.min(bulletAge,age);break;}}
                var gate=WeaponPlatform.indexedGun(id.toString());
                String line=id+" gunAge="+gate.age()+" family="+family+" bullets@"+(bulletAge==99?"never":bulletAge)+" magazines("+mags+")@"+(magAge==99?"never":magAge);
                log("QA_MAG {}",line);
                if(family==null)continue;
                if(bulletAge==99)noBullet++;
                if(mags==0){noMag++;neverList.add(line);}
                else if(magAge==99){never++;neverList.add(line);}
                else if(bulletAge!=99&&magAge>bulletAge){later++;laterList.add(line);}
                else if(magAge==bulletAge)same++;else earlier++;
            }
            log("QA_MAG_SUMMARY guns={} withFamily={} noMagazineListed={} magazineNeverUnlocks={} magazineLaterThanBullet={} sameAge={} magazineEarlier={} noBulletListed={}",guns,inSystem,noMag,never,later,same,earlier,noBullet);
            for(var l:laterList)log("QA_MAG_LATER {}",l);for(var l:neverList)log("QA_MAG_NEVER {}",l);
            sp().getInventory().clearContent();scanned=true;
        }),"scan every gun");
        until("the scan is done",20*60*10,()->scanned,()->{});
        step(20,()->{},"done");
    }
}
