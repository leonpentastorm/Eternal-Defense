    /** The count cache agrees with a full scan of the 270 slots. */
    static boolean cacheMatches(HubStorage s){

    }

    static boolean reloading;static int reloadTicks;
    @SubscribeEvent public static void onTick(TickEvent.ClientTickEvent e){
        if(e.phase!=TickEvent.Phase.END)return;var mc=mc();
        if(!created){
            if(mc.screen instanceof TitleScreen&&mc.getOverlay()==null&&++tick>60){
                created=true;var settings=new LevelSettings("qa_hub",GameType.CREATIVE,false,Difficulty.PEACEFUL,true,new GameRules(),WorldDataConfiguration.DEFAULT);
                mc.createWorldOpenFlows().createFreshLevel("qa_hub",settings,new WorldOptions(40L,false,false),access->access.registryOrThrow(Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions());
            }
            return;
        }
        if(reloading){   // the world was saved and closed: open it again from the title screen
            if(mc.screen instanceof TitleScreen&&mc.getOverlay()==null&&++reloadTicks>40){reloading=false;log("QA_NOTE reopening the world");mc.createWorldOpenFlows().loadLevel(new TitleScreen(),"qa_hub");}
            return;
        }
        if(mc.player==null||mc.level==null||mc.getSingleplayerServer()==null)return;
        if(!built){if(++tick<200)return;built=true;sx=(int)Math.floor(mc.player.getX());sy=(int)Math.floor(mc.player.getY());sz=(int)Math.floor(mc.player.getZ());places();script();}
        if(holding)return;
        if(wait>0){wait--;return;}
        var head=STEPS.peek();
        if(head!=null&&head.length>3){boolean ok;try{ok=((java.util.function.BooleanSupplier)head[3]).getAsBoolean();}catch(Throwable t){ok=false;}int[] rem=(int[])head[4];if(!ok&&rem[0]-->0){wait=3;return;}if(!ok)log("QA_TIMEOUT {}",head[2]);}
        var step=STEPS.poll();
        if(step==null){log("QA_DONE pass={} fail={}",pass,fail);mc.stop();return;}
        log("QA_STEP {}",step[2]);
        try{((Runnable)step[1]).run();}catch(Throwable t){fail++;com.mojang.logging.LogUtils.getLogger().error("QA_STEP_ERROR {}",step[2],t);}
        wait=(int)step[0];
    }

                pack(new ItemStack(ArsenalBeacon.LOGISTICS_HUB_ITEM.get()));sp().getInventory().selected=0;});mc().player.getInventory().selected=0;},"creative break");
        step(5,()->click(HB.below(2),Direction.UP),"place a new, empty Hub B");
        step(30,()->srv(()->sp().gameMode.destroyBlock(HB.offset(-1,-1,-1))),"a survival player breaks one of its parts (it is empty)");
        step(10,()->{String r=srvGet(()->{int items=0;for(var e:sl().getEntitiesOfClass(ItemEntity.class,new AABB(HB).inflate(4)))if(e.getItem().is(ArsenalBeacon.LOGISTICS_HUB_ITEM.get()))items+=e.getItem().getCount();
                int left=0;for(var p:LogisticsHub.cells(HB))if(LogisticsHub.isHub(sl().getBlockState(p)))left++;return items+" "+left;});
            check("an empty Hub breaks whole and drops its item once",r.equals("1 0"),r);
            srv(()->{for(var e:sl().getEntitiesOfClass(ItemEntity.class,new AABB(HB).inflate(4)))e.discard();});},"empty break");

        // ---- unloaded chunks: the Hub's four chunks unload, a force-loaded part chunk alone still takes items, then save, quit and reload ---------------------
        final String[] before={""};
        step(5,()->{srv(()->{var h=hub(A);h.storage.drain();for(int i=0;i<200;i++)h.storage.insert(st(i%2==0?Items.COBBLESTONE:Items.ANDESITE,64),false);h.storage.insert(new ItemStack(Items.SHIELD),false);
                var named=new ItemStack(Items.DIAMOND_SWORD);named.setHoverName(Component.literal("QA Blade"));h.storage.insert(named,false);});
            before[0]=srvGet(()->sum(hub(A)));tp(sx+.5,sy,sz+.5,0,10);},"Hub A holds 202 stacks (one named sword); walk back to spawn, 2000 blocks away");
        until("Hub A's chunks unload",900,()->srvGet(()->{for(var p:LogisticsHub.cells(A))if(sl().hasChunkAt(p))return false;return true;}),()->{});
        step(5,()->check("with all four of its chunks unloaded, the Hub's inventory is intact in the saved data",srvGet(()->{var h=HubData.get(sl()).all().stream().filter(x->x.anchor.equals(A)).findFirst().orElse(null);return h!=null&&sum(h).equals(before[0]);}),""),"unloaded");
        // one chunk that holds only parts, loaded on its own (a ticket of level 33: the chunk itself, not its neighbours): the anchor's chunk stays unloaded
        final net.minecraft.world.level.ChunkPos part=new net.minecraft.world.level.ChunkPos(A.offset(-1,0,-1));
        final net.minecraft.server.level.TicketType<net.minecraft.world.level.ChunkPos> QA_TICKET=net.minecraft.server.level.TicketType.create("qa_hub",Comparator.comparingLong(net.minecraft.world.level.ChunkPos::toLong));
        step(5,()->srv(()->sl().getChunkSource().addRegionTicket(QA_TICKET,part,0,part)),"load only the chunk of the Hub's north-west parts");
        until("that chunk loads",100,()->srvGet(()->sl().hasChunkAt(A.offset(-1,0,-1))),()->{});
        step(5,()->{
            check("the part's chunk is loaded, the anchor's is not",srvGet(()->sl().hasChunkAt(A.offset(-1,0,-1))&&!sl().hasChunkAt(A)),"");
            String r=srvGet(()->{var be=sl().getBlockEntity(A.offset(-1,1,-1));if(be==null)return "no block entity";
                var cap=be.getCapability(ForgeCapabilities.ITEM_HANDLER,Direction.UP).resolve().orElse(null);
                if(cap==null){var h=HubData.get(sl()).all().stream().filter(x->x.anchor.equals(A)).findFirst().orElse(null);var d=CampaignData.get(sl());
                    return "no handler: hub "+(h==null?"none":h.placed+" working "+h.working(sl()))+", cell id "+(be instanceof LogisticsHub.Cell c?c.hubId():"?")+", record id "+(h==null?"?":h.id)+", beacon chunk "+sl().hasChunkAt(d.beacon)+", installed "+d.installed()+", inside "+d.inside(A);}
                var rest=net.minecraftforge.items.ItemHandlerHelper.insertItemStacked(cap,st(Items.GOLD_INGOT,8),false);
                var out=cap.extractItem(0,64,false);
                var h=HubData.get(sl()).all().stream().filter(x->x.anchor.equals(A)).findFirst().orElse(null);
                return rest.getCount()+" "+out.getCount()+" "+(h==null?-1:h.storage.count(Items.GOLD_INGOT))+" "+(h==null?-1:h.storage.count(Items.COBBLESTONE)+h.storage.count(Items.ANDESITE))+" "+(h!=null&&cacheMatches(h.storage));});
            check("its part hands out the one inventory: 8 gold in, a stack of 64 out, the rest untouched",r.equals("0 64 8 "+(200*64-64)+" true"),r);
            srv(()->sl().getChunkSource().removeRegionTicket(QA_TICKET,part,0,part));
            before[0]=srvGet(()->sum(HubData.get(sl()).all().stream().filter(x->x.anchor.equals(A)).findFirst().get()));},"used through a part while the anchor is unloaded");
        step(60,()->{},"let the ticket go");
        step(40,()->{log("QA_NOTE saving and quitting");CHAT.clear();
            boolean local=mc().isLocalServer();mc().level.disconnect();if(local)mc().clearLevel(new net.minecraft.client.gui.screens.GenericDirtMessageScreen(Component.literal("Saving")));else mc().clearLevel();
            mc().setScreen(new TitleScreen());reloading=true;reloadTicks=0;},"save and quit to the title screen");
        step(100,()->{},"the world is open again; let it settle at spawn");
        step(5,()->{check("after the reload, with the Hub's chunks not loaded, its inventory is exactly as saved",srvGet(()->{var h=HubData.get(sl()).all().stream().filter(x->x.anchor.equals(A)).findFirst().orElse(null);return !sl().hasChunkAt(A)&&h!=null&&h.placed&&sum(h).equals(before[0]);}),"");
            srv(()->{sp().teleportTo(sl(),A.getX()+.5,sy,A.getZ()+3.5,180,20);sp().getInventory().clearContent();});},"reloaded; back to the Hub (its south side)");
        until("the Hub's and the beacon's chunks load",300,()->loaded(A)&&loaded(B),()->{});
        step(40,()->{hold(ItemStack.EMPTY);},"settle");
        step(5,()->click(A.offset(-1,-1,1),Direction.SOUTH),"open it from a part");
        step(15,()->{var s=hubScreen();check("its screen opens and reads 202 slots in use",s!=null&&s.getMenu().used()==202,s==null?String.valueOf(mc().screen):""+s.getMenu().used());
            check("the named sword kept its name",srvGet(()->{var h=hub(A);for(int i=0;i<HubStorage.SLOTS;i++)if(h.storage.get(i).is(Items.DIAMOND_SWORD))return h.storage.get(i).getHoverName().getString().equals("QA Blade");return false;}),"");
            check("still the same Hub: the anchor's id is the record's",srvGet(()->{var h=hub(A);return h!=null&&sum(h).equals(before[0]);}),"");
            shot("hub12_after_reload");closeAll();},"after the reload");
    }
