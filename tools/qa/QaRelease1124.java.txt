        step(5,()->{hold(new ItemStack(edItem("shaft_dynamo")));near(P0);},"hold a Shaft Dynamo");

        // 4. Each right-click turns it 45 degrees; 8 clicks go all the way round; a sneaking click turns it back
        step(5,()->{hold(ItemStack.EMPTY);CHAT.clear();from(S,Direction.NORTH);},"empty hand, north of it");
        for(int i=1;i<=8;i++){int n=i;
            step(5,()->click(S,Direction.NORTH),"right-click "+n);
            step(6,()->{int want=(2+n)%8;var on=expected(S,want);
                check("click "+n+": faces "+WAYS[want]+", its glow moved with it",srvGet(()->rot(S)==want&&held(S).equals(on)&&on.stream().allMatch(QaRelease1124::glowAt)&&glows(S,25)==on.size()),srvGet(()->rot(S)+" "+held(S)+" all "+glows(S,25)));
                check("click "+n+": it says which way it faces",last().contains("on, facing "+WAYS[want]+"."),last());},"check "+n);
        }
        step(5,()->sneak(true),"sneak");
        step(5,()->click(S,Direction.NORTH),"sneaking right-click");
        step(6,()->{sneak(false);check("a sneaking click turns it back (north-east)",srvGet(()->rot(S))==1&&said("facing north-east"),srvGet(()->rot(S))+" "+last());},"turned back");
        step(60,()->{srv(()->sl().setDayTime(18000));look(S.getX()-6.5,sy+4,S.getZ()+5.5,S.getX()+8,sy,S.getZ()-8);},"midnight, behind and beside it; 3 s for the night to reach the client");
        step(20,()->shot("r1124_spotlight_northeast"),"picture: facing north-east at night");
        step(5,()->from(S,Direction.NORTH),"back beside it (a click reaches only a few blocks)");
        step(5,()->click(S,Direction.NORTH),"turn it to east");
        step(5,()->click(S,Direction.NORTH),"and south-east");
        step(40,()->{check("turned to south-east",srvGet(()->rot(S))==3,"");look(S.getX()-5.5,sy+5,S.getZ()-4.5,S.getX()+8,sy,S.getZ()+8);},"behind it");

            check("only the ground before the wall is lit (3 and 7 blocks out)",got.equals(Set.of(S.east(3),S.east(7)))&&srvGet(()->glows(S,25))==2,""+got);

        step(5,()->{check("the beam reaches out again",srvGet(()->held(S).equals(expected(S,2))),""+srvGet(()->held(S)));hold(new ItemStack(Items.STONE));from(S.east(7),Direction.SOUTH);},"a stone in hand, beside a glow");

        step(60,()->{},"its next check");
        step(5,()->{check("the stone takes the glow's place",srvGet(()->state(S.east(7)).is(Blocks.STONE)),""+srvGet(()->state(S.east(7))));

        step(5,()->srv(()->{if(sl().getBlockEntity(M0) instanceof CreativeMotorBlockEntity m)m.generatedSpeed.setValue(0);}),"the motor stops");

        step(10,()->check("lit again, the beam is back",srvGet(()->held(S).equals(expected(S,3))&&expected(S,3).stream().allMatch(QaRelease1124::glowAt)),""+srvGet(()->held(S))),"back on");

        step(5,()->{srv(()->sl().setDayTime(6000));hold(new ItemStack(EternalDefense.SPOTLIGHT.get()));look(S.getX()-1.5,sy+1,S.getZ()+3.5,S.getX()+.5,sy,S.getZ()+.5);},"noon; looking north-east at the empty spot with a spotlight in hand");

        step(10,()->click(S.below(),Direction.UP),"place it");
        step(60,()->{check("placed north-east",srvGet(()->rot(S))==1,"");hold(ItemStack.EMPTY);look(S.getX()+.5,sy+1,S.getZ()+4.5,S.getX()+.5,sy+1.2,S.getZ()+.5);},"in front");

    }
