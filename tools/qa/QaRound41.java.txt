    static void hookSounds(){
        if(hooked||mc().getConnection()==null)return;hooked=true;
        mc().getConnection().getConnection().channel().pipeline().addBefore("packet_handler","qa_sounds",new io.netty.channel.ChannelInboundHandlerAdapter(){
            @Override public void channelRead(io.netty.channel.ChannelHandlerContext ctx,Object msg)throws Exception{
                ResourceLocation id=null;
                if(msg instanceof net.minecraft.network.protocol.game.ClientboundSoundPacket p)id=p.getSound().value().getLocation();
                else if(msg instanceof net.minecraft.network.protocol.game.ClientboundSoundEntityPacket p)id=p.getSound().value().getLocation();
                if(id!=null&&id.getNamespace().equals(EternalDefense.ID)){SOUNDS.add(id.getPath());log("QA_SOUND_PACKET {}",id);}
                super.channelRead(ctx,msg);
            }
        });
        log("QA_NOTE sound packets are counted at the client's connection");
    }

    @SubscribeEvent public static void onTick(TickEvent.ClientTickEvent e){
        if(e.phase!=TickEvent.Phase.END)return;var mc=mc();
        if(!created){
            if(mc.screen instanceof TitleScreen&&mc.getOverlay()==null&&++tick>60){
                created=true;var settings=new LevelSettings("qa_r41",GameType.CREATIVE,false,Difficulty.PEACEFUL,true,new GameRules(),WorldDataConfiguration.DEFAULT);
                mc.createWorldOpenFlows().createFreshLevel("qa_r41",settings,new WorldOptions(41L,false,false),access->access.registryOrThrow(Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions());
            }
            return;
        }
        if(mc.player==null||mc.level==null||mc.getSingleplayerServer()==null)return;
        if(!built){if(++tick<200)return;built=true;hookSounds();sx=(int)Math.floor(mc.player.getX());sy=(int)Math.floor(mc.player.getY());sz=(int)Math.floor(mc.player.getZ());places();script();}
        if(holding)return;
        if(wait>0){wait--;return;}
        var head=STEPS.peek();
        if(head!=null&&head.length>3){boolean ok;try{ok=((java.util.function.BooleanSupplier)head[3]).getAsBoolean();}catch(Throwable t){ok=false;}int[] rem=(int[])head[4];if(!ok&&rem[0]-->0){wait=3;return;}if(!ok)log("QA_TIMEOUT {}",head[2]);}
        var step=STEPS.poll();
        if(step==null){log("QA_DONE pass={} fail={}",pass,fail);mc.stop();return;}
        log("QA_STEP {}",step[2]);
        try{((Runnable)step[1]).run();}catch(Throwable t){fail++;com.mojang.logging.LogUtils.getLogger().error("QA_STEP_ERROR {}",step[2],t);}
        wait=(int)step[0];
    }

        step(2,()->srv(()->{SupportChatter.say(sl(),sp().position(),SupportCalls.Kind.FIRE,sp());SupportChatter.say(sl(),sp().position(),SupportCalls.Kind.FIRE,sp());SupportChatter.say(sl(),sp().position(),SupportCalls.Kind.SUPPLY,sp());}),"three flare calls answered at once");
        after(20,()->check("three calls at once: one radio line plays",sounds("flare.")==1,String.valueOf(SOUNDS)),"one line");
        step(2,()->srv(()->SupplyCatapult.radio(sp())),"a Specialty Seller order while that line still plays");
        after(20,()->check("an order while a flare line plays adds no voice",sounds("seller.order_chatter")==0,String.valueOf(SOUNDS)),"no order line");
        step(110,()->{},"wait for the line to end (at most 4.75 s)");
        step(2,()->srv(()->{SupplyCatapult.radio(sp());SupplyCatapult.radio(sp());}),"two orders at once after it ended");
        after(20,()->check("after the line ended, the next order is heard, once",sounds("seller.order_chatter")==1&&sounds("flare.")==1,String.valueOf(SOUNDS)),"one order line");
        step(110,()->{},"wait again");
        step(2,()->srv(()->SupportChatter.say(sl(),sp().position(),SupportCalls.Kind.RETURN,sp())),"a return flare call");
        after(20,()->check("and a flare line plays again",sounds("flare.recall_chatter")==1,String.valueOf(SOUNDS)),"line again");

        // An ammo and an attachment table with every part in a later Age (the screen fed with that state)
        for(String kind:List.of("attachment","ammo")){
            for(int locked:new int[]{3,1}){
                step(5,()->{var n=new CompoundTag();n.putString("kind",kind);n.putInt("age",1);for(String k:List.of("gun","ammo","attachment","armor"))n.putInt(k+"Age",k.equals(kind)?1:0);
                    n.putInt("locked",locked);n.putString("notice","Upgrade your "+kind+" table");n.put("recipes",new ListTag());n.putInt("total",0);n.putInt("page",0);n.putInt("pageSize",6);n.put("upgrades",new ListTag());n.put("typeCounts",new CompoundTag());
                    PlatformScreen.receive(new WeaponPlatform.State(n,true,""));recordFrame=true;},"a "+kind+" table, a gun in hand, "+locked+" part(s) in later Ages");
                step(5,()->{String want=kind.equals("ammo")?(locked==1?"There is 1 more ammo type for this weapon in a later Age":"There are 3 more ammo types for this weapon in later Ages"):(locked==1?"There is 1 more attachment for this weapon in a later Age":"There are 3 more attachments for this weapon in later Ages");
                    check(kind+" table, "+locked+" locked: the list says how many wait in later Ages, not Hold a gun",drawnHas(want.substring(0,30))&&!drawnHas("Hold a gun"),drawn());shot("r41_10_"+kind+"_later_"+locked);},"the hint");
                step(5,()->closeAll(),"close");
            }
        }
        step(5,()->{var n=new CompoundTag();n.putString("kind","attachment");n.putInt("age",1);n.putInt("locked",0);n.putString("notice","Equip a gun");n.put("recipes",new ListTag());n.put("upgrades",new ListTag());n.put("typeCounts",new CompoundTag());
            PlatformScreen.receive(new WeaponPlatform.State(n,true,""));recordFrame=true;},"no gun in hand");
        step(5,()->{check("no gun in hand: the old line, Hold a gun",drawnHas("Hold a gun"),drawn());shot("r41_11_attachment_no_gun");},"hold a gun");
        step(5,()->closeAll(),"close");

        // The MRE Station's press, and Create's goggles on every animated block
        step(5,()->srv(()->{sl().setBlock(MRE,EternalDefense.MRE_STATION.get().defaultBlockState(),3);
            var goggles=BuiltInRegistries.ITEM.get(new ResourceLocation("create","goggles"));sp().setItemSlot(EquipmentSlot.HEAD,new ItemStack(goggles));note=String.valueOf(goggles);}),"an MRE Station; Create's goggles on");
        step(5,()->{look(MRE.offset(0,0,-3),MRE.getX()+.5,MRE.getY()+.9,MRE.getZ()+.5);ccStart=classCasts();},"look at the MRE Station");
        step(60,()->shot("r41_12_mre_press"),"the press (the rod now reaches into the hopper)");
        step(5,()->{hold(new ItemStack(EternalDefense.SATELLITE_BEACON_ITEM.get()));tp(SAT.getX()+.5,sy,SAT.getZ()-3.5,0,30);},"hold a Satellite Beacon");
        step(5,()->click(SAT.below(),Direction.UP),"place it");
        step(20,()->{check("the satellite stands",srvGet(()->sl().getBlockState(SAT).is(EternalDefense.SATELLITE_BEACON.get())),srvGet(()->sl().getBlockState(SAT).toString()));
            hold(ItemStack.EMPTY);look(SAT.offset(0,0,-3),SAT.getX()+.5,SAT.getY()+1,SAT.getZ()+.5);},"look at the satellite with the goggles on");
        step(1200,()->shot("r41_13_satellite_goggles"),"a minute of the satellite turning under the goggles");
        step(5,()->{look(A.offset(0,-1,-5),A.getX()+.5,A.getY(),A.getZ()+.5);},"look at the Hub");
        step(200,()->{},"ten seconds");
        step(5,()->look(HALL.offset(0,0,-3),HALL.getX()+.5,HALL.getY()+.5,HALL.getZ()+.5),"look at the Mess Hall");
        step(200,()->{long cc=classCasts()-ccStart;check("goggles on the MRE Station, the satellite (a minute), the Hub and the Mess Hall: no ClassCastException in the log, the game runs",cc==0&&mc().player!=null,"new ClassCastException lines: "+cc+", goggles="+note);},"goggles");

        // Side quest: the mod's installations add base score, the whole set one tier
        step(5,()->{String r=srvGet(()->{
                var blocks=new HashMap<Long,net.minecraft.world.level.block.state.BlockState>();int x=0;
                for(String id:BaseScoring.INSTALLATIONS){String reg=id.equals("mess_hall")||id.equals("drink_mixer")?id+"_mk1":id;
                    var block=BuiltInRegistries.BLOCK.get(new ResourceLocation(EternalDefense.ID,reg));if(block==net.minecraft.world.level.block.Blocks.AIR)return "missing block "+reg;
                    blocks.put(new BlockPos(100000+x++*4,0,100000).asLong(),block.defaultBlockState());}
                var d=CampaignData.get(sl());int all=BaseScoring.analyze(d,blocks,Set.of()).installations();
                // A second Mess Hall (Mk III) and the blocks that are not in the set change nothing
                for(String extra:List.of("mess_hall_mk3","fob_beacon","supply_catapult","supply_delivery_gate","logistics_hub_part"))blocks.put(new BlockPos(100000+x++*4,0,100000).asLong(),BuiltInRegistries.BLOCK.get(new ResourceLocation(EternalDefense.ID,extra)).defaultBlockState());
                int extras=BaseScoring.analyze(d,blocks,Set.of()).installations();
                blocks.entrySet().removeIf(e->e.getValue().is(EternalDefense.COMMAND_TABLE.get()));int missingOne=BaseScoring.analyze(d,blocks,Set.of()).installations();
                return all+" "+extras+" "+missingOne;});
            check("the scorer, with the game's own blocks: all 20 installations 500 points; a second Mess Hall and FOB-only blocks add nothing; 19 give 475",r.equals("500 500 475"),r);},"the installations, counted by the scorer");
        step(5,()->srv(()->BaseSurvey.invalidate(sl())),"survey the base again (the Hub, the Mess Hall, the MRE Station and the satellite stand in it)");
        step(60,()->{},"the survey starts");
        until("the survey is done",2400,()->srvGet(()->{var s=BaseSurvey.get(sl());return !s.scanning&&s.score!=null;}),()->{});
        step(5,()->sync(),"sync");
        step(10,()->{int points=srvGet(()->BaseSurvey.get(sl()).score.installations());
            check("the live survey: 4 installations in the zone, 100 points, in the total",points==100,points+" points, total "+srvGet(()->BaseSurvey.get(sl()).score.total()));
            var s=new BeaconClient.ControlScreen();s.tab=0;mc().setScreen(s);recordFrame=true;},"open the Overview");
        step(5,()->{check("the Overview's base score shows Installations 100 (4/20) and the Total",drawnHas("Installations")&&drawnHas("100 (4/20)")&&drawnHas("Total"),drawn());shot("r41_14_overview_installations");},"Overview, installations");
        step(5,()->closeAll(),"close");
    }
