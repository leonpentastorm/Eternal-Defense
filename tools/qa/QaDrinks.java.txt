    /** No text over another, no text across a button's edge (a label painted inside its own button is fine), everything in the panel; cut texts fail when strict. */
    static void layout(String name,boolean strict){
        var s=mc().screen;var boxes=new ArrayList<>(new LinkedHashSet<>(lastFrame));var bad=new ArrayList<String>();var cut=new ArrayList<String>();
        String size=(s==null?0:s.width)+"x"+(s==null?0:s.height);
        for(int i=0;i<boxes.size();i++){var a=boxes.get(i);
            if(a.cut())cut.add(a.text());
            if(s!=null&&(a.x()<0||a.y()<0||a.x()+a.w()>s.width||a.y()+a.h()>s.height))bad.add("off the window: '"+a.text()+"'");
            for(int j=i+1;j<boxes.size();j++){var b=boxes.get(j);if(hit(a.x(),a.y(),a.w(),a.h(),b.x(),b.y(),b.w(),b.h()))bad.add("'"+a.text()+"' over '"+b.text()+"'");}
            if(s!=null)for(var c:s.children())if(c instanceof net.minecraft.client.gui.components.AbstractWidget w&&w.visible&&hit(a.x(),a.y(),a.w(),a.h(),w.getX(),w.getY(),w.getWidth(),w.getHeight())&&!inside(a.x(),a.y(),a.w(),a.h(),w.getX(),w.getY(),w.getWidth(),w.getHeight()))bad.add("'"+a.text()+"' across the button '"+w.getMessage().getString()+"'");
        }
        if(s instanceof net.minecraft.client.gui.screens.inventory.AbstractContainerScreen<?> cs){int px=cs.getGuiLeft(),py=cs.getGuiTop(),pw=cs.getXSize(),ph=cs.getYSize();
            for(var a:boxes)if(a.x()<px||a.y()<py||a.x()+a.w()>px+pw||a.y()+a.h()>py+ph)bad.add("outside the panel: '"+a.text()+"'");
            for(var c:s.children())if(c instanceof net.minecraft.client.gui.components.AbstractWidget w&&w.visible&&(w.getX()<px||w.getY()<py||w.getX()+w.getWidth()>px+pw||w.getY()+w.getHeight()>py+ph))bad.add("the button '"+w.getMessage().getString()+"' outside the panel");
            // the widgets do not overlap one another either
            var ws=new ArrayList<net.minecraft.client.gui.components.AbstractWidget>();for(var c:s.children())if(c instanceof net.minecraft.client.gui.components.AbstractWidget w&&w.visible)ws.add(w);
            for(int i=0;i<ws.size();i++)for(int j=i+1;j<ws.size();j++){var a=ws.get(i);var b=ws.get(j);if(hit(a.getX(),a.getY(),a.getWidth(),a.getHeight(),b.getX(),b.getY(),b.getWidth(),b.getHeight()))bad.add("button over button");}
            // and no visible slot sits under a button
            for(var slot:cs.getMenu().slots)if(slot.isActive())for(var w:ws)if(hit(px+slot.x,py+slot.y,16,16,w.getX(),w.getY(),w.getWidth(),w.getHeight()))bad.add("a slot under the button '"+w.getMessage().getString()+"'");
        }
        if(s!=null)for(var c:s.children())if(c instanceof Ui.UiButton b&&b.visible&&b.label&&!b.getMessage().getString().isEmpty()){int room=b.look==Ui.Look.ROW?b.getWidth()-16:b.getWidth()-10;if(mc().font.width(b.getMessage())>room)cut.add("button '"+b.getMessage().getString()+"'");}
        check(name+" at "+size+": "+boxes.size()+" texts, none over another or across a button, all in the panel",bad.isEmpty()&&!boxes.isEmpty(),bad.size()+" "+String.join(" | ",bad.subList(0,Math.min(5,bad.size()))));
        if(strict)check(name+" at "+size+": no text cut short",cut.isEmpty(),String.join(" | ",cut));
        else if(!cut.isEmpty())log("QA_NOTE {} at {}: cut short: {}",name,size,String.join(" | ",cut));
    }

    }

    }

    @SubscribeEvent public static void onTick(TickEvent.ClientTickEvent e){
        if(e.phase!=TickEvent.Phase.END)return;var mc=mc();
        if(!created){
            if(mc.screen instanceof TitleScreen&&mc.getOverlay()==null&&++tick>60){
                created=true;var settings=new LevelSettings("qa_drinks",GameType.CREATIVE,false,Difficulty.NORMAL,true,new GameRules(),WorldDataConfiguration.DEFAULT);
                mc.createWorldOpenFlows().createFreshLevel("qa_drinks",settings,new WorldOptions(77L,false,false),access->access.registryOrThrow(Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.FLAT).value().createWorldDimensions());
            }
            return;
        }
        if(mc.player==null||mc.level==null||mc.getSingleplayerServer()==null)return;
        if(!built){if(++tick<200)return;built=true;sx=(int)Math.floor(mc.player.getX());sy=(int)Math.floor(mc.player.getY());sz=(int)Math.floor(mc.player.getZ());script();}
        if(holding)return;
        if(wait>0){wait--;return;}
        var head=STEPS.peek();
        if(head!=null&&head.length>3){boolean ok;try{ok=((java.util.function.BooleanSupplier)head[3]).getAsBoolean();}catch(Throwable t){ok=false;}int[] rem=(int[])head[4];if(!ok&&rem[0]-->0){wait=3;return;}if(!ok)log("QA_TIMEOUT {}",head[2]);}
        var step=STEPS.poll();
        if(step==null){log("QA_DONE pass={} fail={}",pass,fail);mc.stop();return;}
        try{((Runnable)step[1]).run();}catch(Throwable t){fail++;com.mojang.logging.LogUtils.getLogger().error("QA_STEP_ERROR {}",step[2],t);}
        wait=(int)step[0];
    }

        step(5,()->srv(()->{sp().teleportTo(sl(),sx+.5,sy,sz+4.5,180,10);PlayerMeals.eat(sp(),new MealData(true,List.of(new MealData.Bonus(MealRules.Effect.HARDY,1),new MealData.Bonus(MealRules.Effect.BLAST_WARD,1),new MealData.Bonus(MealRules.Effect.PROJECTILE_WARD,1)),MealRules.FIELD_TICKS));}),"eat Hardy, Blast Ward and Projectile Ward");
        step(10,()->{
            check("the three show as their own effects",mc().player.hasEffect(MealEffects.of(MealRules.Effect.HARDY))&&mc().player.hasEffect(MealEffects.of(MealRules.Effect.BLAST_WARD))&&mc().player.hasEffect(MealEffects.of(MealRules.Effect.PROJECTILE_WARD)),"");
            check("with their names",MealEffects.of(MealRules.Effect.HARDY).getDisplayName().getString().equals("Hardy")&&MealEffects.of(MealRules.Effect.BLAST_WARD).getDisplayName().getString().equals("Blast Ward")&&MealEffects.of(MealRules.Effect.PROJECTILE_WARD).getDisplayName().getString().equals("Projectile Ward"),"");
            check("and their icons",mc().getResourceManager().getResource(MealEffects.icon(MealRules.Effect.HARDY)).isPresent()&&mc().getResourceManager().getResource(MealEffects.icon(MealRules.Effect.BLAST_WARD)).isPresent()&&mc().getResourceManager().getResource(MealEffects.icon(MealRules.Effect.PROJECTILE_WARD)).isPresent(),"");
            float blast=srvGet(()->net.minecraftforge.common.ForgeHooks.onLivingHurt(sp(),sl().damageSources().explosion(null,null),10f));
            float arrow=srvGet(()->net.minecraftforge.common.ForgeHooks.onLivingHurt(sp(),new net.minecraft.world.damagesource.DamageSource(sl().registryAccess().registryOrThrow(Registries.DAMAGE_TYPE).getHolderOrThrow(net.minecraft.world.damagesource.DamageTypes.ARROW)),10f));
            float fire=srvGet(()->net.minecraftforge.common.ForgeHooks.onLivingHurt(sp(),sl().damageSources().inFire(),10f));
            check("Blast Ward at home: 40 percent off explosions",Math.abs(blast-6f)<1e-3,""+blast);check("Projectile Ward at home: 30 percent off arrows",Math.abs(arrow-7f)<1e-3,""+arrow);check("no fire protection any more",fire==10f,""+fire);
        },"meal effects");
        step(5,()->mc().setScreen(new net.minecraft.client.gui.screens.inventory.InventoryScreen(mc().player)),"the inventory");
        step(10,()->shot("d18_meal_effects"),"meal effects shot");
        step(3,()->mc().setScreen(null),"close");
        step(5,()->srv(()->{sl().setBlock(HALL(),ArsenalBeacon.MESS_HALL_IV.get().defaultBlockState(),3);sp().teleportTo(sl(),sx-9.5,sy,sz+3.5,180,10);}),"a Mess Hall Mk IV, and stand in front of it");
        step(10,()->srv(()->{var h=(MessHall.HallEntity)sl().getBlockEntity(HALL());net.minecraftforge.network.NetworkHooks.openScreen(sp(),h,b->{b.writeBlockPos(HALL());b.writeVarInt(h.mk());});}),"open it");
        until("the hall",60,()->mc().screen instanceof MessHallScreen,()->{});
        step(10,()->{var s=(MessHallScreen)mc().screen;var c=s.cells.stream().filter(x->x.effect==MealRules.Effect.HARDY).findFirst().orElseThrow();hover(c.getX()+10,c.getY()+10);},"hover Hardy");
        step(8,()->shot("d19_hall_hardy"),"Hardy's recipe tooltip");
        step(3,()->{hover(0,0);close();},"close");

        // ---- the Field Guide page and Workshop Fabrication ------------------------------------------------------------------------------
        step(5,()->{var g=new BeaconClient.GuideScreen(null);mc().setScreen(g);},"open the Field Guide");
        step(10,()->{if(mc().screen instanceof BeaconClient.GuideScreen g){g.page=Arrays.asList(GuideText.IDS).indexOf("drinks");g.scroll=0;}},"the drinks page");
        step(30,()->{check("the Field Guide has the drinks page",mc().screen instanceof BeaconClient.GuideScreen g&&GuideText.IDS[g.page].equals("drinks"),"");shot("d20_guide_drinks");},"guide");
        step(5,()->{var s=new BeaconClient.ControlScreen();s.tab=2;mc().setScreen(s);},"open Workshop Fabrication");
        step(40,()->{shot("d21_workshop");},"workshop");
        step(3,()->mc().setScreen(null),"close");

        // ---- death clears the drink ---------------------------------------------------------------------------------------------------
        step(3,()->srv(()->PlayerDrinks.drink(sp(),new DrinkData(List.of(new DrinkData.Part(new net.minecraft.resources.ResourceLocation("speed"),0)),12000,false))),"a drink");
        step(5,()->srv(()->sp().kill()),"die");
        step(20,()->{check("death clears the drink",srvGet(()->PlayerDrinks.of(sp())==null),"");},"dead");
        step(5,()->{if(mc().player!=null)mc().player.respawn();},"respawn");
        step(20,()->{},"settle");
    }
