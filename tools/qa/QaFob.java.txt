    static void layout(String name,boolean strict){
        var s=mc().screen;var boxes=new ArrayList<>(new LinkedHashSet<>(lastFrame));var bad=new ArrayList<String>();var cut=new ArrayList<String>();
        String size=(s==null?0:s.width)+"x"+(s==null?0:s.height);
        for(int i=0;i<boxes.size();i++){var a=boxes.get(i);
            if(a.cut())cut.add(a.text());
            if(s!=null&&(a.x()<0||a.y()<0||a.x()+a.w()>s.width||a.y()+a.h()>s.height))bad.add("off the window: '"+a.text()+"'");
            for(int j=i+1;j<boxes.size();j++){var b=boxes.get(j);if(hit(a.x(),a.y(),a.w(),a.h(),b.x(),b.y(),b.w(),b.h()))bad.add("'"+a.text()+"' over '"+b.text()+"'");}
            if(s!=null)for(var c:s.children())if(c instanceof net.minecraft.client.gui.components.AbstractWidget w&&w.visible&&hit(a.x(),a.y(),a.w(),a.h(),w.getX(),w.getY(),w.getWidth(),w.getHeight()))bad.add("'"+a.text()+"' over the button '"+w.getMessage().getString()+"'");
        }
        // every text and button inside the screen's panel
        if(s instanceof net.minecraft.client.gui.screens.inventory.AbstractContainerScreen<?> cs){int px=cs.getGuiLeft(),py=cs.getGuiTop(),pw=cs.getXSize(),ph=cs.getYSize();
            for(var a:boxes)if(a.x()<px||a.y()<py||a.x()+a.w()>px+pw||a.y()+a.h()>py+ph)bad.add("outside the panel: '"+a.text()+"'");
            for(var c:s.children())if(c instanceof net.minecraft.client.gui.components.AbstractWidget w&&w.visible&&(w.getX()<px||w.getY()<py||w.getX()+w.getWidth()>px+pw||w.getY()+w.getHeight()>py+ph))bad.add("the button '"+w.getMessage().getString()+"' outside the panel");}
        // a button's label room, as Ui.UiButton draws it: a row 16 px less than its width, a switch less its knob and its ON/OFF, any other 10 px less
        if(s!=null)for(var c:s.children())if(c instanceof Ui.UiButton b&&b.visible&&b.label&&!b.getMessage().getString().isEmpty()){
            int room=b.look==Ui.Look.ROW?b.getWidth()-16:b.look==Ui.Look.TOGGLE?b.getWidth()-30-mc().font.width(Ui.t("off"))-24:b.getWidth()-10-(b.locked?10:0);
            if(mc().font.width(b.getMessage())>room)cut.add("button '"+b.getMessage().getString()+"'");}
        check(name+" at "+size+": "+boxes.size()+" texts, none over another or over a button, all in the panel",bad.isEmpty()&&!boxes.isEmpty(),bad.size()+" "+String.join(" | ",bad.subList(0,Math.min(4,bad.size()))));
        if(strict)check(name+" at "+size+": no text cut short",cut.isEmpty(),String.join(" | ",cut));
        else if(!cut.isEmpty())log("QA_NOTE {} at {}: cut short: {}",name,size,String.join(" | ",cut));
    }

    @SubscribeEvent public static void onTick(TickEvent.ClientTickEvent e){
        if(e.phase!=TickEvent.Phase.END)return;var mc=mc();
        if(!created){
            if(mc.screen instanceof TitleScreen&&mc.getOverlay()==null&&++tick>60){
                created=true;
                var settings=new LevelSettings("qa_fob",GameType.CREATIVE,false,Difficulty.NORMAL,true,new GameRules(),WorldDataConfiguration.DEFAULT);
                mc.createWorldOpenFlows().createFreshLevel("qa_fob",settings,new WorldOptions(20201010L,true,false),access->access.registryOrThrow(Registries.WORLD_PRESET).getHolderOrThrow(WorldPresets.NORMAL).value().createWorldDimensions());
            }
            return;
        }
        if(mc.player==null||mc.level==null||mc.getSingleplayerServer()==null)return;
        if(!started){
            if(++tick<300)return;started=true;
            cmd("gamerule doMobSpawning false");cmd("gamerule doDaylightCycle false");cmd("weather clear");cmd("time set 6000");
            srv(()->{MinecraftForge.EVENT_BUS.register(new FobGameTests.Runner());FobGameTests.Runner.start(sl());});
            return;
        }
        if(!testsDone){
            if(FobGameTests.Runner.lastPass<0)return;
            testsDone=true;check("FOB GameTests",FobGameTests.Runner.lastFail==0,"pass="+FobGameTests.Runner.lastPass+" fail="+FobGameTests.Runner.lastFail);
            srv(FobGameTests.Runner::release);
            script();return;
        }
        if(wait>0){wait--;return;}
        var head=STEPS.peek();
        if(head!=null&&head.length>3){boolean ok;try{ok=((java.util.function.BooleanSupplier)head[3]).getAsBoolean();}catch(Throwable t){ok=false;}int[] rem=(int[])head[4];if(!ok&&rem[0]-->0){wait=3;return;}if(!ok){fail++;log("QA_CHECK FAIL timeout {}","" + head[2]);}}
        var step=STEPS.poll();
        if(step==null){log("QA_DONE pass={} fail={}",pass,fail);mc.stop();return;}
        try{((Runnable)step[1]).run();}catch(Throwable t){fail++;com.mojang.logging.LogUtils.getLogger().error("QA_STEP_ERROR {}",step[2],t);}
        wait=(int)step[0];
    }

     * 0.0.30, the owner's 4-wide gate across two rails along X, seen from the south (so it faces east): lane A is the returned cell, lane B the
     * cell south of it, the anchor north of lane A and the left column south of lane B. The spot needs sturdy ground under all four cells and
     * under the lanes 2 blocks either way, the gate's 4 x 2 free, and the anchor inside the FOB zone ({@code fob}) or the base zone.
     */
    static BlockPos gateSpot(BlockPos around,int reach,boolean fob){
        var l=sl();BlockPos best=null;double bestD=1e9;
        for(int dx=-reach;dx<=reach;dx++)for(int dz=-reach;dz<=reach;dz++){
            int x=around.getX()+dx,z=around.getZ()+dz,y=l.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES,x,z);var r=new BlockPos(x,y,z);
            boolean free=true;
            for(int side=-1;side<=2&&free;side++){var c=r.relative(Direction.SOUTH,side);
                if(!l.getBlockState(c.below()).isFaceSturdy(l,c.below(),Direction.UP))free=false;
                for(int up=0;up<=1&&free;up++)if(!l.getBlockState(c.above(up)).isAir())free=false;}
            for(int a=-2;a<=2&&free;a++)for(int side=0;side<=1&&free;side++){var c=r.offset(a,0,side);if(!l.getBlockState(c).isAir()||!l.getBlockState(c.below()).isFaceSturdy(l,c.below(),Direction.UP))free=false;}
            if(!free||Math.abs(dx)<3&&Math.abs(dz)<4||!fob&&Math.abs(dz)<6)continue;   // at home: off to one side, so the beacon's beam is not behind it
            if(fob?Fobs.zoneAt(l,r.north())==null:BaseZone.problem(l,r.north())!=null)continue;
            double d=dx*dx+dz*dz;if(d<bestD){best=r;bestD=d;}
        }
        return best;
    }

    static BlockPos findSpot(){
        var l=sl();var from=sp().blockPosition();
        if(FobZone.sections(l,from).isEmpty())return null;   // not a village (yet) where the player stands
        var candidates=new ArrayList<BlockPos>();
        for(int dx=-20;dx<=20;dx++)for(int dz=-20;dz<=20;dz++){
            int x=from.getX()+dx,z=from.getZ()+dz,y=l.getHeight(Heightmap.Types.MOTION_BLOCKING_NO_LEAVES,x,z);var at=new BlockPos(x,y,z);
            if(!l.getBlockState(at.below()).isFaceSturdy(l,at.below(),Direction.UP)||!l.getBlockState(at).isAir()||!l.getBlockState(at.above()).isAir())continue;
            var side=at.relative(Direction.SOUTH,3);if(!l.getBlockState(side).isAir()||!l.getBlockState(side.above()).isAir())continue;
            if(!l.getEntitiesOfClass(net.minecraft.world.entity.Entity.class,new net.minecraft.world.phys.AABB(at).inflate(1)).isEmpty())continue;
            candidates.add(at);
        }
        candidates.sort(Comparator.comparingDouble(p->p.distSqr(from)));
        for(var at:candidates)if(Fobs.plantProblem(l,at)==null)return at;
        return null;
    }
